       ③ 用了扣什么（代价）

   ── 定义形状（契约给）──
     { id, 名,
       先天: false,                        ← true 就谁也学不会（血继限界那种）
       条件: [ {类型, ……} ],               ← 施展时才检查
       代价: { 资源字段: 量 } }            ← 用了就扣

   条件有四种类型（够用了）：
     { 类型: '有状态',  状态: 'id' }      要在某个状态里才能放
     { 类型: '没状态',  状态: 'id' }      带着某个状态时放不出
     { 类型: '资源够',  字段: 'x', 至少: N }
     { 类型: '字段够',  字段: 'x', 至少: N }
   ============================================================ */

this.能学吗 = function (定义) {
  var _d = 定义 || {};
  if (_d.先天) return { 可以: false, 原因: '这个学不了，是天生的' };
  return { 可以: true };
};

this.能用吗 = function (定义, 值表, 身上) {
  var _d = 定义 || {};
  var _v = 值表 || {};
  var _身 = 身上 || {};
  var _条 = _d.条件 || [];
  var _缺 = [];
  for (var _i = 0; _i < _条.length; _i++) {
    var _c = _条[_i];
    var _t = _c.类型;
    if (_t === '有状态') {
      if (!_身[_c.状态]) _缺.push('要处在「' + _c.状态 + '」里才能用');
    } else if (_t === '没状态') {
      if (_身[_c.状态]) _缺.push('带着「' + _c.状态 + '」时用不了');
    } else if (_t === '资源够' || _t === '字段够') {
      var _现 = Number(_v[_c.字段]) || 0;
      var _要 = Number(_c.至少) || 0;
      if (_现 < _要) _缺.push(_c.字段 + ' 还差 ' + (_要 - _现));
    }
  }
  return { 可以: _缺.length === 0, 缺: _缺 };
};

/* 用了之后扣什么（★ 扣不动就别放 —— 先问再用） */
this.用得起吗 = function (定义, 值表) {
  var _代 = (定义 || {}).代价 || {};
  var _v = 值表 || {};
  var _缺 = [];
  for (var _k in _代) {
    var _现 = Number(_v[_k]) || 0;
    var _要 = Number(_代[_k]) || 0;
    if (_现 < _要) _缺.push({ 字段: _k, 现: _现, 需: _要 });
  }
  return { 可以: _缺.length === 0, 缺: _缺 };
};

/* 一次问全：能不能用 + 用不用得起。★ 两关都过才允许放 */
this.放得出吗 = function (定义, 值表, 身上) {
  var _能 = this.能用吗(定义, 值表, 身上);
  if (!_能.可以) return { 可以: false, 因为: '条件不到', 缺: _能.缺 };
  var _付 = this.用得起吗(定义, 值表);
  if (!_付.可以) return { 可以: false, 因为: '付不起代价', 缺: _付.缺.map(function (x) { return x.字段 + ' 还差 ' + (x.需 - x.现); }) };

/* 扣代价 */

/* 一次做完：检查 → 扣代价 → 进冷却（冷却走 E13 的状态机制） */
this.施展 = function (定义, 值表, 身上, 冷却回合) {

  if (!_r.可以) return _r;
  this.付代价(定义, 值表);
  if (冷却回合 && this.进冷却) this.进冷却(身上, (定义 || {}).id, 冷却回合);
  return { 可以: true, 放了: (定义 || {}).id };

/* ════════════════════════════════════════════════════════════
   C6-b · 改图（图不是静态的，玩家能自己改）

   ★ 参考自 色色灵感状态栏V3.738 的天赋树实现（见 参考-色色灵感状态栏.md）：
     它的树是**玩家花代价改的**，而且改完要记次数 —— 用户说的
     「还能自定义生成」指的就是这个。

   三件事必须一起做，缺一条就不对：
     ① 先校验（改完还成不成图）—— 不校验就改，图会坏
     ② 再算代价（加节点 / 改门槛 / 删节点 各要多少）
     ③ 记住改了几次（次数本身就是数据：改得越多越贵，或作为统计）

   ★★ 第四件最关键：**保存失败要退费**。
     不能出现「代价扣了但图没改成功」—— 那是事务，不是算术。
     本实现的做法：先把改动算在一个**副本**上，成功了才落盘；
     落了盘才算成功，扣费在落盘之后。顺序反了就会出现「白扣钱」。
   ============================================================ */

/* 先校验：这个改动合不合法（不会把图改坏） */
this.校验改动 = function (图, 改动) {
  var _n = (图 || {}).节点 || [];
  var _d = 改动 || {};
  var _有 = {};
  for (var _i = 0; _i < _n.length; _i++) _有[_n[_i].id] = _n[_i];

  if (_d.动作 === '加节点') {
    if (!_d.节点 || !_d.节点.id) return { 合法: false, 原因: '没给节点' };
    if (_有[_d.节点.id]) return { 合法: false, 原因: '这个 id 已经有了' };
    var _前 = _d.节点.前置 || [];
    for (var _j = 0; _j < _前.length; _j++) {
      if (!_有[_前[_j]] && _前[_j] !== _d.节点.id) return { 合法: false, 原因: '前置不存在：' + _前[_j] };
    }
    return { 合法: true };
  }

  if (_d.动作 === '删节点') {
    if (!_有[_d.id]) return { 合法: false, 原因: '没有这个节点' };
    /* ★ 删之前先看有没有人依赖它 —— 直接删会留下悬空的前置 */
    var _靠它的 = [];
    for (var _k = 0; _k < _n.length; _k++) {
      var _前2 = _n[_k].前置 || [];
      if (_前2.indexOf(_d.id) >= 0) _靠它的.push(_n[_k].id);
    }
    if (_靠它的.length) return { 合法: false, 原因: '还有东西依赖它', 依赖: _靠它的 };
    return { 合法: true };
  }

  if (_d.动作 === '改门槛') {
    if (!_有[_d.id]) return { 合法: false, 原因: '没有这个节点' };
    return { 合法: true };
  }
  return { 合法: false, 原因: '不认这个动作：' + _d.动作 };
};

/* 算代价（代价表由契约给）。★ 次数越多越贵 */
this.算改动代价 = function (改动, 代价表, 已改次数) {
  var _c = 代价表 || {};
  var _d = 改动 || {};
  var _基 = (_c.动作 || {})[_d.动作] || {};
  var _次 = Number(已改次数) || 0;
  var _涨 = Number(_c.每次递增) || 0;
  var _倍 = 1 + _涨 * _次;
  var _出 = {};
  for (var _k in _基) _出[_k] = Math.ceil((Number(_基[_k]) || 0) * _倍);
  return _出;
};

/* 代价支付得起吗（复用 C3 的口径） */
this.改动付得起吗 = function (代价, 值表) {
  return this.canAfford(代价, 值表);
};

/* 落盘：把改动真正写进图（在副本上做，成功才替换） */
this.落图 = function (图, 改动) {
  var _d = 改动 || {};
  /* ★ 深拷一份改，改完再交出去 —— 这样中途出错原图不受影响 */
  var _新 = JSON.parse(JSON.stringify(图 || { 节点: [] }));
  var _n = _新.节点 || (_新.节点 = []);
  if (_d.动作 === '加节点') {
    _n.push(_d.节点);
    return { 可以: true, 图: _新 };
  }
  if (_d.动作 === '删节点') {
    for (var _i = 0; _i < _n.length; _i++) if (_n[_i].id === _d.id) { _n.splice(_i, 1); break; }
    return { 可以: true, 图: _新 };
  }
  if (_d.动作 === '改门槛') {
    for (var _j = 0; _j < _n.length; _j++) {
      if (_n[_j].id === _d.id) { _n[_j].需要 = _d.需要 || _n[_j].需要; break; }
    }
    return { 可以: true, 图: _新 };
  }
  return { 可以: false, 原因: '不认这个动作' };
};

/* ★★ 一次性做完，顺序不能错：
     校验 → 算代价 → 付得起吗 → **改副本** → **落盘成功** → 才扣代价
   反过来的话会出现「钱扣了但图没改成功」—— 那是事务错，不是算术错 */
this.改图 = function (图, 改动, 代价表, 值表, 已改次数) {
  var _校 = this.校验改动(图, 改动);
  if (!_校.合法) return { 可以: false, 在哪一步: '校验', 原因: _校.原因, 依赖: _校.依赖 };

  var _代 = this.算改动代价(改动, 代价表, 已改次数);
  var _付 = this.canAfford(_代, 值表);
  if (!_付.可以) return { 可以: false, 在哪一步: '付代价', 缺: _付.缺, 要: _代 };

  /* ★ 先在副本上落盘。这一步失败就当什么都没发生 */
  var _落 = this.落图(图, 改动);
  if (!_落.可以) return { 可以: false, 在哪一步: '落盘', 原因: _落.原因 };

  /* 落盘成功了才扣费 */
  for (var _k in _代) 值表[_k] = (Number(值表[_k]) || 0) - (Number(_代[_k]) || 0);

  /* ★ 落盘之后自检一次：图坏了就把费用退回去（退费是这条链的兜底） */
  var _环 = this.有没有环 ? this.有没有环(_落.图) : { 有环: false };
  if (_环.有环) {
    for (var _k2 in _代) 值表[_k2] = (Number(值表[_k2]) || 0) + (Number(_代[_k2]) || 0);
    return { 可以: false, 在哪一步: '自检', 原因: '改完出现环，费用已退回', 环在: _环.在哪 };
  }

  return { 可以: true, 图: _落.图, 花了: _代, 改了几次: (Number(已改次数) || 0) + 1 };
};
