/* ════════════════════════════════════════════════════════════
   ★ 命中率：战斗引擎**自带**，不依赖任何外部文件。

   原来这里用的是成长引擎里的 rateFromDiff —— 那是**跨引擎依赖**，
   违反「引擎自包含」（同一类问题在 NSFW 那次已经吃过一次）。

   但**这里是唯一实现**：战斗只看得到自己这一份。
   ============================================================ */
this.命中率 = function (diff, cfg) {
  var _c = cfg || {};
  var _基 = Number(_c.基准);
  if (isNaN(_基)) _基 = 45;
  var _斜 = Number(_c.斜率);
  if (isNaN(_斜)) _斜 = 2.2;
  var _下 = Number(_c.下限);
  if (isNaN(_下)) _下 = 1;
  var _上 = Number(_c.上限);
  if (isNaN(_上)) _上 = 95;
  var _v = _基 + (Number(diff) || 0) * _斜;
  return Math.max(_下, Math.min(_上, _v));
};

/* 兼容：外面传了 rate 就用外面的，没传就用自带的 */
this.取命中算法 = function (rate) {
  if (typeof rate === 'function') return rate;
  var _自 = this;
  return function (diff, cfg) { return _自.命中率(diff, cfg); };
};

this.strike = function (攻方, 守方, diff, rng, rateFromDiff) {
  if (typeof rateFromDiff !== 'function') rateFromDiff = this.取命中算法(null);
  var _命中率 = rateFromDiff(diff + (Number(攻方.命中加成) || 0) - (Number(守方.闪避加成) || 0), 攻方.命中);
  var _掷 = rng() * 100;
  if (_掷 >= _命中率) {
    return { 命中: false, 掷值: 掷(_掷), 命中率: _命中率, 伤害: 0 };
  }
  var _基础 = (Number(攻方.攻击) || 0) - (Number(守方.防御) || 0);
  var _伤害 = Math.max(1, Math.round(_基础 * (Number(攻方.伤害倍率) || 1)));
  return { 命中: true, 掷值: 掷(_掷), 命中率: _命中率, 伤害: _伤害 };
};

function 掷(v) { return Math.round(v * 100) / 100; }

/* ════════════════════════════════════════════════════════════

function clone(o) {

/* ════════════════════════════════════════════════════════════
   三、摘要（★ 这是给 AI 的唯一出口）

   纪律：**不写掷值、不写命中率、不写公式**。
        写的是「谁赢了、多快、伤成什么样、有没有动用杀招、有没有留下不可逆的伤」。

this.受伤档 = function (剩余, 上限) {
  var _p = (Number(上限) || 1) <= 0 ? 0 : (Number(剩余) || 0) / Number(上限) * 100;
  if (_p <= 0) return '倒下';
  if (_p < 20) return '濒死';
  if (_p < 50) return '重伤';
  if (_p < 80) return '轻伤';

};

this.summarize = function (r, cfg) {

  if (r.胜者 === '我方' || r.胜者 === '对方') {

    else if (_差 <= 8) _结果 = r.胜者 + '·常规';

  } else if (r.胜者 === '未决') {

  }

  var _不可逆 = [];
  var _我掉 = (Number(r.原我方.有效值) || 100) - (Number(r.我方.有效值) || 100);
  var _敌掉 = (Number(r.原对方.有效值) || 100) - (Number(r.对方.有效值) || 100);
  if (_我掉 > 0) _不可逆.push('我方受伤 ' + Math.round(_我掉) + '%');
  if (_敌掉 > 0) _不可逆.push('对方受伤 ' + Math.round(_敌掉) + '%');

  return {
    结果: _结果,
    回合数: r.回合数,
    我方状态: _我p + '（剩 ' + Math.round(r.我方.生命) + '/' + Math.round(Number(r.原我方.上限) || 0) + '）',
    对方状态: _敌p + '（剩 ' + Math.round(r.对方.生命) + '/' + Math.round(Number(r.原对方.上限) || 0) + '）',

function 写法约束(结果, 我伤, 敌伤) {

  if (String(结果).indexOf('苦战') >= 0) _r.push('必须写出耗与险，双方都到过危险处');
  if (String(结果).indexOf('僵持') >= 0) _r.push('以未分胜负收场，谁也没占到便宜，留出下一次');
  if (String(结果).indexOf('同归') >= 0) _r.push('两败俱伤，双方都倒下');
  if (敌伤 === '倒下') _r.push('对方倒下这一下要写实，不能一笔带过');
  if (我伤 === '濒死' || 我伤 === '倒下') _r.push('我方到了极限，身体上的代价必须落在正文里');

/* ════════════════════════════════════════════════════════════

   谁需要它：

this.multi = function (cfg, rng, rateFromDiff) {

  var _对方 = (_c.对方 || []).slice();

  var _策略 = _c.选目标 || '打最弱的';   /* 默认打最弱：先减员，这是最常见的打法 */
  var _群伤摊薄 = Number(_c.群体摊薄) || 0.6;
  var _日志 = [];
  var _回合 = 0;

  function 活着(a) {
    var _n = 0;
    for (var _i = 0; _i < a.length; _i++) if (a[_i].生命 > 0) _n++;
    return _n;
  }

  /* ★ 选目标：这是策略，不是随机。写死随机会让同一局面每次重渲染死不同的人 */
  function 挑(候选) {

    for (var _i = 0; _i < 候选.length; _i++) if (候选[_i].生命 > 0) _活.push(候选[_i]);

    } else if (_策略 === '打血最厚的') {
      _活.sort(function (x, y) { return (Number(y.生命) || 0) - (Number(x.生命) || 0); });
    } else if (_策略 === '按顺序') {
      /* 不动，按给的次序 */
    } else {
      /* 打最弱的（默认）—— 先减员 */
      _活.sort(function (x, y) { return (Number(x.生命) || 0) - (Number(y.生命) || 0); });
    }

  }

  }

  if (_结果.indexOf('碾压') >= 0) _约束.push('快而不费力，过程短，不必写拉锯');
  if (_结果.indexOf('苦战') >= 0) _约束.push('必须写出耗与险，双方都到过危险处');
  if (_结果.indexOf('僵持') >= 0 || _胜者 === '未决') _约束.push('以未分胜负收场，留出下一次');
