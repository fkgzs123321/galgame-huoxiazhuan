/* 矿脉发现判定: 基于采矿技能与biome修正, 使用DDL骰子引擎判定
   返回是否发现/矿脉类型/矿脉数据/检定结果 */
function _veinDiscover(skill, biome) {
  var skl = Number(skill) || 0;
  var b = String(biome) || '温带森林';
  var bMod = BIOME_MINING_MOD[b] || BIOME_MINING_MOD['温带森林'];
  var score = skl * 5 + 20;
  var seed = 'vein_' + b + '_' + skl;
  var checkResult = _check(score, seed, '普通');
  var discovered = checkResult.result === '普通成功' || checkResult.result === '极难成功' || checkResult.result === '大成功';
  var veinType = null;
  var veinData = null;
  if (discovered) {
    var bonus = 0;
    if (checkResult.result === '大成功') bonus = 0.15;
    else if (checkResult.result === '极难成功') bonus = 0.05;
    var rollSeed = 'veintype_' + seed + '_' + checkResult.roll;
    var typeRoll = _lcgRoll(rollSeed, 1, 100)[0] / 100;
    var cumulative = 0;
    var discoverMod = bMod.发现修正 + bonus;
    for (var k in VEIN_DB) {
      var v = VEIN_DB[k];
      var adjProb = v.发现概率 * discoverMod;
      cumulative += adjProb;
      if (typeRoll <= cumulative / (cumulative + (1 - cumulative) * 0.5)) {
        veinType = k;
        veinData = v;
        break;
      }
    }
    if (!veinType) {
      veinType = '小矿脉';
      veinData = VEIN_DB['小矿脉'];
    }
  }
  return {
    发现: discovered,
    矿脉类型: veinType,
    矿脉数据: veinData,
    biome修正: bMod.发现修正,
    检定: checkResult
  };
}

/* 挖掘速度: 实际速度 = 基础速度 x (1 + 技能 x 0.06) / 硬度修正
   硬度越高速度越慢, 技能越高速度越快
   返回每小时产出单位数 */
function _miningSpeed(skill, mineralHardness) {
  var skl = Number(skill) || 0;
  var hard = Number(mineralHardness) || 5;
  if (hard < 1) hard = 1;
  var base = 10;
  var speed = base * (1 + skl * 0.06) / (hard / 5);
  return Math.round(speed * 100) / 100;
}

/* 挖掘产出: 产出数量 = 矿脉数量 x (0.6 + 技能 x 0.04) x 品质修正
   技能越高产出越多, 大成功额外+20%
   返回实际产出/产出系数/检定结果 */
function _miningYield(skill, veinSize, seed) {
  var skl = Number(skill) || 0;
  var size = Number(veinSize) || 10;
  var score = skl * 5 + 10;
  var sd = String(seed) || 'mine_yield';
  var checkResult = _check(score, sd, '普通');
  var yieldMod = 0.6 + skl * 0.04;
  var bonus = 0;
  if (checkResult.result === '大成功') { yieldMod *= 1.2; bonus = 0.2; }
  else if (checkResult.result === '大失败') { yieldMod *= 0.6; bonus = -0.4; }
  else if (checkResult.result === '失败') { yieldMod *= 0.8; bonus = -0.2; }
  var actual = Math.max(1, Math.round(size * yieldMod));
  return {
    实际产出: actual,
    矿脉数量: size,
    产出系数: Math.round(yieldMod * 100) / 100,
    附加修正: bonus,
    检定: checkResult
  };
}

/* 深层采矿: 深度越大难度越高, 需要更高技能与设备
   返回可达深度/速度修正/可用矿物列表/检定结果 */
function _deepMining(skill, depth) {
  var skl = Number(skill) || 0;
  var dep = Number(depth) || 0;
  var score = skl * 4 + 10;
  var sd = 'deepmine_' + skl + '_' + dep;
  var checkResult = _check(score, sd, '困难');
  var maxDepth = skl * 8 + 20;
  var speedMod = 1.0;
  if (dep > maxDepth) speedMod = 0.3;
  else if (dep > maxDepth * 0.7) speedMod = 0.6;
  else if (dep > maxDepth * 0.5) speedMod = 0.8;
  var available = [];
  for (var k in DEEP_MINERAL_DB) {
    var dm = DEEP_MINERAL_DB[k];
    if (dep >= dm.深度要求 && dep <= maxDepth) {
      available.push(k);
    }
  }
  var success = checkResult.result === '普通成功' || checkResult.result === '极难成功' || checkResult.result === '大成功';
  return {
    可达深度: maxDepth,
    速度修正: speedMod,
    可用矿物: available,
    深度足够: dep <= maxDepth,
    挖掘成功: success,
    检定: checkResult
  };
}

    建议: advice
  };
}
