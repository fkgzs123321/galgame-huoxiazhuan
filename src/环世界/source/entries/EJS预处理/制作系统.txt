/* 制作技能判定: 根据技能类型返回有效技能值与修正
   类型映射: 手工/裁缝/锻造 -> 手工技能; 雕刻/艺术 -> 艺术品技能
   注: DDL中_craftSkill(skill,reqSkill)负责成功率/品质,此处_craftTypeMod负责类型修正
   公式以DDL骰子引擎为准,禁止本地修改 */
function _craftTypeMod(skill, type) {
  var skl = Number(skill) || 0;
  var t = String(type) || '手工';
  var baseMod;
  if (t === '手工') { baseMod = 1.0; }
  else if (t === '裁缝') { baseMod = 1.0; }
  else if (t === '锻造') { baseMod = 0.95; }

  else if (t === '艺术') { baseMod = 1.2; }
  else { baseMod = 1.0; }
  var effective = Math.round(skl * baseMod * 100) / 100;
  return {
    原始技能: skl,
    类型: t,
    类型修正: baseMod,
    有效技能: effective
  };
}

/* 制作耗时公式: 实际时间 = 基础耗时 / (制作台速度 x (1 + 有效技能 x 0.08))
   公式以DDL骰子引擎为准,禁止本地修改 */
function _craftTime(base, skill, station) {
  var b = Number(base) || 1;
  var skl = Number(skill) || 0;
  var st = String(station) || '手工台';
  var stData = CRAFT_STATION_DB[st] || CRAFT_STATION_DB['手工台'];
  var speed = Number(stData.速度) || 0.8;
  var actual = b / (speed * (1 + skl * 0.08));
  return Math.round(actual * 100) / 100;
}

/* 品质判定: 基于技能+制作台修正,使用LCG骰子(DDL引擎)滚动百分比
   返回品质等级/价值系数/美观修正/检定结果 */
function _qualityCheck(skill, station, seed) {
  var skl = Number(skill) || 0;
  var st = String(station) || '手工台';
  var stData = CRAFT_STATION_DB[st] || CRAFT_STATION_DB['手工台'];
  var stationSpeed = Number(stData.速度) || 0.8;
  var skillScore = skl * 5 + Math.round(stationSpeed * 10);
  var checkResult = _check(skillScore, seed, '普通');
  var roll = checkResult.roll;
  var quality;
  var i;
  for (i = 0; i < QUALITY_DB.length; i++) {
    var lo = Number(QUALITY_DB[i].区间[0]) || 0;
    var hi = Number(QUALITY_DB[i].区间[1]) || 100;
    if (roll >= lo && roll <= hi) {
      quality = QUALITY_DB[i];
      break;
    }
  }
  if (!quality) { quality = QUALITY_DB[1]; }
  var bonus = 0;
  if (checkResult.result === '大成功') {
    for (i = 0; i < QUALITY_DB.length; i++) {
      if (QUALITY_DB[i].等级 === '优秀') { quality = QUALITY_DB[i]; break; }
    }
    bonus = 10;
  } else if (checkResult.result === '大失败') {
    quality = QUALITY_DB[0];
    bonus = -5;
  }
  return {
    骰值: roll,
    品质: quality.等级,
    价值系数: quality.价值,
    美观修正: quality.美观 + bonus,
    检定: checkResult,
    附加修正: bonus
  };
}

/* 产出计算: 产出数量 = 基础产出 x (1 + 有效技能 x 0.05) x 品质系数(良好及以上加成)
   公式以DDL骰子引擎为准,禁止本地修改 */
function _craftYield(base, skill, quality) {
  var b = Number(base) || 1;
  var skl = Number(skill) || 0;
  var q = String(quality) || '普通';
  var yieldMod = 1 + skl * 0.05;
  if (q === '糟糕') { yieldMod *= 0.8; }
  else if (q === '普通') { yieldMod *= 1.0; }

  else if (q === '大师') { yieldMod *= 1.3; }
  else if (q === '传奇') { yieldMod *= 1.5; }
  return Math.max(1, Math.round(b * yieldMod * 100) / 100);
}
