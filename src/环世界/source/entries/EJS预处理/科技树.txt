/* 研究速度公式: 研究点产出 = 研究技能 x 0.5 x 设备修正 x 专注修正 / 小时 */
function _researchSpeed(skill, deviceMod, focusMod) {
  var skl = Number(skill) || 0;
  var dm = Number(deviceMod) || 0;
  var fm = Number(focusMod) || 1;
  var output = skl * 0.5 * dm * fm;
  return Math.round(output * 100) / 100;
}

/* 设备修正: 无台0 / 简单研究台1.0 / 高科技研究台1.5(需高科技) / 多维研究台3.0 */
function _deviceMod(deviceType) {
  if (deviceType === 'none') { return { mod: 0, note: '无研究台(无法研究)' }; }
  if (deviceType === 'simple') { return { mod: 1.0, note: '简单研究台 x1.0' }; }
  if (deviceType === 'advanced') { return { mod: 1.5, note: '高科技研究台 x1.5(需高科技)' }; }
  if (deviceType === 'multi') { return { mod: 3.0, note: '多维研究台 x3.0' }; }
  return { mod: 0, note: '无研究台' };
}

/* 专注修正: 单人1.0 / 多人协助 = 1.0 + 协助者数 x 0.3 (最多2协助) */
function _focusMod(helperCount) {
  var hc = Number(helperCount) || 0;
  if (hc <= 0) { return { mod: 1.0, note: '单人研究 x1.0' }; }
  var cappedHelpers = Math.min(hc, 2);
  var bonus = cappedHelpers * 0.3;
  return { mod: Math.round((1.0 + bonus) * 100) / 100, note: '多人协助 x' + (1.0 + bonus) };
}
