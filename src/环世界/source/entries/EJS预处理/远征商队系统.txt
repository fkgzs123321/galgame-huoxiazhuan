/* 旅行时间计算
   distance(km), terrain(地形名), speed(基础速度km/h), load(当前载重kg), maxLoad(最大载重kg)
   公式: distance / (speed × terrainMod × (1 - load/maxLoad × 0.3))
   负重越接近上限速度越慢(最多减速30%) */
function _travelTime(distance, terrain, speed, load, maxLoad) {
  var dist = Number(distance) || 0;
  var spd = Number(speed) || 1;
  var tMod = TERRAIN_SPEED_DB[String(terrain)] || 1.0;
  var ld = Math.max(0, Number(load) || 0);
  var ml = Math.max(1, Number(maxLoad) || 1);
  var loadRatio = Math.min(1, ld / ml);
  var speedFactor = 1 - loadRatio * 0.3;
  var effectiveSpeed = spd * tMod * speedFactor;
  var time = effectiveSpeed > 0 ? dist / effectiveSpeed : 9999;
  return Math.round(time * 100) / 100;
}

/* 商队有效速度(基础速度 + 驮兽速度加成,按动物类型计一次)
   caravanType(商队类型), animals=[{type,count}]
   返回: { base, animalBonus, effective } */
function _caravanSpeed(caravanType, animals) {
  var cfg = CARAVAN_DB[String(caravanType)] || CARAVAN_DB['贸易队'];
  var base = Number(cfg.speed) || 4;
  var seenTypes = {};
  var bonus = 0;
  if (animals) {
    for (var i = 0; i < animals.length; i++) {
      var t = String(animals[i].type);
      var info = PACK_ANIMAL_DB[t];
      if (info && !seenTypes[t]) {
        bonus += info.speedMod;
        seenTypes[t] = true;
      }
    }
  }
  return { base: base, animalBonus: bonus, effective: Math.round((base + bonus) * 100) / 100 };
}

/* 商队遇袭判定
   seed(骰子种子), distance(km), factionRelation(派系关系值)
   距离>10km时5%基础概率,派系敌对(关系<-10)+20% */
function _caravanEncounter(seed, distance, factionRelation) {
  var dist = Number(distance) || 0;
  var rel = Number(factionRelation) || 0;
  var baseProb = dist > 10 ? 5 : 0;
  var hostile = rel < -10;
  var totalProb = baseProb + (hostile ? 20 : 0);
  totalProb = Math.max(0, Math.min(95, totalProb));
  var check = _check(totalProb, String(seed), '困难');
  return {
    遇袭: (totalProb > 0 && check.result !== '失败' && check.result !== '大失败'),
    概率: totalProb,
    骰值: check.roll,
    阈值: check.threshold,
    派系敌对: hostile,
    结果: check.result
  };
}

/* 驮兽总载重计算
   animals=[{type, count}, ...]
   返回: 总载重kg */
function _packAnimalLoad(animals) {
  var total = 0;
  if (!animals) { return 0; }
  for (var i = 0; i < animals.length; i++) {
    var a = animals[i];
    var info = PACK_ANIMAL_DB[String(a.type)];
    if (info) {
      var count = Number(a.count) || 1;
      total += info.load * count;
    }
  }
  return total;
}

/* 口粮消耗计算
   colonists(殖民者数), animals=[{type,count}], duration(h)
   公式: colonists×0.5N/h + 各驮兽×各自消耗 */
function _provisionConsumption(colonists, animals, duration) {
  var col = Number(colonists) || 0;
  var dur = Number(duration) || 0;
  var total = col * 0.5 * dur;
  if (animals) {
    for (var i = 0; i < animals.length; i++) {
      var info = PACK_ANIMAL_DB[String(animals[i].type)];
      if (info) {
        var count = Number(animals[i].count) || 1;
        total += info.food * count * dur;
      }
    }
  }
  return Math.round(total * 100) / 100;
}

/* 组建商队
   members(殖民者数或数组), animals=[{type,count,tame}], supplies={type,silver,food,distance,relation}
   验证人数/载重/口粮/驯服度是否足够
   返回: 组建结果(含有效速度/载重/旅行时间/口粮需求/问题列表) */
function _caravanForm(members, animals, supplies) {
  var sp = supplies || {};
  var caravanType = sp.type || '贸易队';
  var config = CARAVAN_DB[caravanType] || CARAVAN_DB['贸易队'];
  var people = Array.isArray(members) ? members.length : (Number(members) || 0);
  var packLoad = _packAnimalLoad(animals);
  var totalCapacity = config.load + packLoad;
  var cargoWeight = (Number(sp.food) || 0) * 0.5;
  var distance = Number(sp.distance) || 0;
  var spd = _caravanSpeed(caravanType, animals);
  var travelTime = _travelTime(distance, biome, spd.effective, cargoWeight, totalCapacity);
  var roundTrip = travelTime * 2;
  var foodNeeded = _provisionConsumption(people, animals, roundTrip);
  var issues = [];
  if (people > config.maxPeople) { issues.push('超员(' + people + '>' + config.maxPeople + ')'); }
  if (cargoWeight > totalCapacity) { issues.push('超载(' + cargoWeight.toFixed(1) + '>' + totalCapacity + ')'); }
  if ((Number(sp.food) || 0) < foodNeeded) { issues.push('口粮不足(' + sp.food + '<' + foodNeeded + 'N)'); }
  if (animals) {
    for (var i = 0; i < animals.length; i++) {
      var info = PACK_ANIMAL_DB[String(animals[i].type)];
      if (info) {
        var tame = Number(animals[i].tame) || 0;
        if (tame < info.tameReq) { issues.push(animals[i].type + '驯服不足(' + tame + '<' + info.tameReq + ')'); }
      }
    }
  }
  return {
    有效: issues.length === 0,
    类型: caravanType,
    人数: people,
    最大人数: config.maxPeople,
    基础速度: spd.base,
    驮兽加成: spd.animalBonus,
    有效速度: spd.effective,
    载重容量: totalCapacity,
    商队载重: config.load,
    驮兽载重: packLoad,
    货物重量: Math.round(cargoWeight * 100) / 100,
    旅行时间: travelTime,
    往返时间: roundTrip,
    需要口粮: foodNeeded,
    携带口粮: Number(sp.food) || 0,
    贸易修正: _caravanTrade(p1Social, Number(sp.relation) || 0, gameDay),
    问题: issues
  };
}

/* 远程贸易成功率
   skill(社交), factionRelation(派系关系), goods(商品标识用于种子)
   公式: 社交×3 + 关系修正(关系值÷10取整)×5 + 50
   返回: 成功率与加成明细 */

  var check = _check(successRate, 'caravan_trade_' + String(goods), '普通');
  return {
    成功率: successRate,
    加成: bonus,
    社交贡献: skl * 3,
    关系贡献: relMod * 5,
    关系修正: relMod,
    骰值: check.roll,
    阈值: check.threshold,
    结果: check.result,
    成功: (check.result !== '失败' && check.result !== '大失败')
  };
}
