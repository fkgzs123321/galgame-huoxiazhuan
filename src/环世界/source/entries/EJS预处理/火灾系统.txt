/* 获取火势规模等级 */

  var s = Number(size) || 0;
  for (var k in FIRE_SIZE_DB) {

/* 火势蔓延概率: fireSize × windSpeed × material系数, 使用DDL骰子引擎判定
   返回是否蔓延/蔓延规模增量/新着火点数量/检定结果 */
function _fireSpread(fireSize, windSpeed, material) {
  var size = Number(fireSize) || 0;
  var wind = Number(windSpeed) || 0;
  var mat = _fireMaterialInfo(material);
  var sizeLevel = _fireSizeLevel(size);
  var spreadRate = mat.蔓延速度 * sizeLevel.数据.蔓延修正;
  var spreadScore = size + wind * 5 * spreadRate;
  var seed = 'firespread_' + material + '_' + size + '_' + wind;
  var checkResult = _check(spreadScore, seed, '普通');
  var willSpread = checkResult.result === '普通成功' || checkResult.result === '极难成功' || checkResult.result === '大成功';
  var sizeIncrement = 0;
  var newIgnitions = 0;
  if (willSpread) {
    var bonus = 0;
    if (checkResult.result === '大成功') bonus = 0.5;
    else if (checkResult.result === '极难成功') bonus = 0.2;
    sizeIncrement = Math.round((size * 0.15 + wind * 2) * spreadRate * (1 + bonus));
    var igniteSeed = 'ignitecount_' + seed + '_' + checkResult.roll;
    var igniteRoll = _lcgRoll(igniteSeed, 1, 100)[0] / 100;
    if (mat.燃点等级 === '极低') newIgnitions = Math.floor(igniteRoll * 3) + 1;
    else if (mat.燃点等级 === '低') newIgnitions = Math.floor(igniteRoll * 2) + 1;
    else newIgnitions = Math.floor(igniteRoll * 1) + 1;
  }
  return {
    是否蔓延: willSpread,
    蔓延规模增量: sizeIncrement,
    新着火点: newIgnitions,
    蔓延系数: Math.round(spreadRate * 100) / 100,
    检定: checkResult
  };
}

/* 燃烧时长计算: material基础时长 × size规模修正
   返回总燃烧时长(小时)/规模修正/是否自然熄灭 */
function _fireDuration(material, size) {

  if (sz > 100) sizeMod = 1.8;
  else if (sz > 60) sizeMod = 1.5;
  else if (sz > 30) sizeMod = 1.2;
  else if (sz < 10) sizeMod = 0.7;

  if (weather === '大雨' || weather === '暴风雪') { total *= 0.4; naturalExtinguish = sz < 30; }
  else if (weather === '雨' || weather === '雪') { total *= 0.7; }
  total = Math.round(total * 100) / 100;
  return {
    总燃烧时长: total,
    规模修正: sizeMod,
    是否自然熄灭: naturalExtinguish,
    天气: weather
  };
}

/* 烟雾影响: density(1-10) × duration(小时), 心情-5~-20, 呼吸损伤

function _smokeEffect(density, duration) {

  if (dens > 10) dens = 10;

  var visionMod = 1.0 - dens * 0.08;
  if (visionMod < 0.2) visionMod = 0.2;
  var severity = '轻微';
  var desc = '略有呛鼻';
  if (dens >= 8) { severity = '严重'; desc = '浓烟呛人,视野极差,呼吸损伤明显'; }

  else if (dens >= 4) { severity = '轻度'; desc = '烟雾可见,略有呛鼻'; }
  return {
    烟雾密度: dens,
    持续时长: dur,
    心情惩罚: moodPenalty,
    呼吸损伤: breathDamage,
    视野修正: Math.round(visionMod * 100) / 100,
    严重度: severity,
    描述: desc
  };

## 火灾规则
- **点燃判定**：闪电(×1.5初始规模20)/营火失控(×1.0初始15)/爆炸引燃(×2.0初始40)
- **温度修正**：>35℃点燃概率×1.4, <0℃点燃概率×0.5
- **火势蔓延**：评分=规模+风速×5×材料蔓延系数, 规模等级影响蔓延修正
- **扑灭速度**：每回合扑灭量=(建设×2+工具加成+泡沫50)×0.5×检定修正
- **工具加成**：徒手0/水桶+20/灭火器+35/泡沫+50
- **灼伤分级**：轻5HP红肿/中15HP水泡/重30HP真皮损伤/致命50HP组织碳化
- **燃烧时长**：基础时长×规模修正(小火0.7~烈火1.8), 大雨降至40%
- **烟雾影响**：密度≥8心情-20呼吸损伤明显, 密度4-7心情-10~-15
- **大雨灭火**：每回合灭火量=雨量×15×(1-规模/200), 大雨可自然扑灭小火
- **不可燃材料**：金属/石头完全不燃烧, 燃点999

## 叙事要求
- 发生"闪电引燃"时，叙事须描写雷击的闪光与焦糊气味
- 火势升级为"大火"时，叙事须描写殖民者的紧迫与热浪扑面
- 发生"烈火"时，叙事须描写撤离的混乱与殖民地建筑的崩塌
- 殖民者遭受"重度灼伤"时，叙事须描写皮肤焦黑与痛苦哀嚎
- 化学燃料火灾时，叙事须描写毒烟的刺鼻与防护紧迫
- 大雨扑灭火灾时，叙事须描写雨水浇灭火势的嘶嘶白雾
- 营火失控时，叙事须描写火苗窜出火坑引燃周围物品的过程
