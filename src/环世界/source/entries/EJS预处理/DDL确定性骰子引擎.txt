function _hashSeed(seed) {
  var s = String(seed);
  var h = 5381;
  for (var i = 0; i < s.length; i++) {
    h = ((h << 5) + h + s.charCodeAt(i)) | 0;
  }
  return Math.abs(h) || 1;
}

function _lcgRoll(seed, count, sides) {
  var a = 1103515245;
  var c = 12345;
  var m = 2147483648;
  var state = _hashSeed(seed) & 0x7FFFFFFF;
  var results = [];
  for (var i = 0; i < count; i++) {
    state = ((Math.imul(a, state) + c) >>> 0) & 0x7FFFFFFF;
    results.push((state % sides) + 1);
  }
  return results;
}
