function _cropMod(skill, temp, light, fertility) {
  var skl = Number(skill) || 0;

  var tempMod;
  if (t >= 16 && t <= 26) { tempMod = 1.0; }
  else if ((t >= 10 && t <= 15) || (t >= 27 && t <= 35)) { tempMod = 0.7; }
  else { tempMod = 0; }

  var lightMod;
  if (light === '充足') { lightMod = 1.0; }
  else if (light === '不足') { lightMod = 0.5; }
  else { lightMod = 0; }

  var fertMod;
  if (fertility === '肥沃') { fertMod = 1.2; }
  else if (fertility === '普通') { fertMod = 1.0; }
  else if (fertility === '贫瘠') { fertMod = 0.6; }
  else { fertMod = 1.0; }

  var growthMod = tempMod * lightMod * fertMod * (1 + skl * 0.06) * (1 + skl * 0.02);
  var yieldMod = (1 + skl * 0.03) * fertMod;
  return {
    生长修正: Math.round(growthMod * 1000) / 1000,
    产量修正: Math.round(yieldMod * 1000) / 1000,
    防治率: Math.min(80, skl * 4) / 100
  };
}
