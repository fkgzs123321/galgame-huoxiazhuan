/* 综合室温计算
   inside=室内基础温度, outside=外部温度, walls=墙体隔热总系数
   heater=暖气总功率°C/h, cooler=冷气总功率°C/h, colonists=室内人数 */
function _roomTemp(inside, outside, walls, heater, cooler, colonists) {
  var wallInsulation = Number(walls) || 0;
  var heatGain = Number(heater) || 0;
  var heatLoss = Number(cooler) || 0;
  var bodyHeat = (Number(colonists) || 0) * 0.5;
  /* 墙体隔热让室内向外部靠拢的速度放缓，温差衰减 = (1 - 隔热系数) */
  var tempDiff = outside - inside;
  var leakedTemp = tempDiff * (1 - Math.min(1, Math.max(0, wallInsulation)));
  var roomTemp = inside + leakedTemp + heatGain - heatLoss + bodyHeat;
  return Math.round(roomTemp * 10) / 10;
}

/* 失温判定 (temp<5°C时每小时HP-1~-5)
   colonist=殖民者对象, temp=暴露温度, duration=暴露小时数 */
function _hypothermia(colonist, temp, duration) {
  var t = Number(temp) || 20;
  var d = Number(duration) || 0;
  if (t >= 5) { return { risk: false, note: '温度安全' }; }
  /* 越冷越严重: 0~5°C -> -1/h, -5~0°C -> -3/h, <-5°C -> -5/h */
  var hpLossPerHour = 0;
  var severity = '轻微';
  if (t >= 0 && t < 5) { hpLossPerHour = 1; severity = '轻微'; }
  else if (t >= -5 && t < 0) { hpLossPerHour = 3; severity = '中度'; }
  else if (t < -5) { hpLossPerHour = 5; severity = '严重'; }
  /* 衣物保温抵消 */
  var insulation = _insulationValue(colonist.衣物);
  var effectiveLoss = Math.max(0, hpLossPerHour - Math.floor(insulation / 5));
  var totalLoss = effectiveLoss * d;
  return {
    risk: true,
    severity: severity,
    hpLossPerHour: effectiveLoss,
    totalHpLoss: totalLoss,
    duration: d,
    note: '失温(' + severity + ') ' + t + '°C 暴露' + d + 'h HP-' + totalLoss
  };
}

/* 中暑判定 (temp>35°C时每小时HP-1~-5)
   colonist=殖民者对象, temp=暴露温度, duration=暴露小时数 */
function _heatstroke(colonist, temp, duration) {
  var t = Number(temp) || 20;
  var d = Number(duration) || 0;
  if (t <= 35) { return { risk: false, note: '温度安全' }; }
  /* 越热越严重: 35~40°C -> -1/h, 40~45°C -> -3/h, >45°C -> -5/h */
  var hpLossPerHour = 0;
  var severity = '轻微';
  if (t > 35 && t <= 40) { hpLossPerHour = 1; severity = '轻微'; }
  else if (t > 40 && t <= 45) { hpLossPerHour = 3; severity = '中度'; }
  else if (t > 45) { hpLossPerHour = 5; severity = '严重'; }
  /* 疲劳影响体温调节 */
  var fatigue = Number(colonist.疲劳) || 0;
  var fatiguePenalty = fatigue > 80 ? 1 : 0;
  var effectiveLoss = hpLossPerHour + fatiguePenalty;
  var totalLoss = effectiveLoss * d;
  return {
    risk: true,
    severity: severity,
    hpLossPerHour: effectiveLoss,
    totalHpLoss: totalLoss,
    duration: d,
    note: '中暑(' + severity + ') ' + t + '°C 暴露' + d + 'h HP-' + totalLoss
  };
}

/* 舒适温度区间 (基于衣物+特质)
   默认 16~28°C，衣物保温值扩展下限，防热特质扩展上限 */
function _comfortTemp(colonist) {
  var lower = 16;
  var upper = 28;
  var insulation = _insulationValue(colonist.衣物);
  /* 每点保温值扩展下限1°C */
  lower = lower - insulation;
  /* 寒冷特质扩展上限+5°C, 耐热特质扩展上限+10°C */
  if (colonist.特质 === '耐热') { upper = upper + 10; }
  if (colonist.特质 === '寒血') { upper = upper + 5; lower = lower - 3; }
  if (colonist.特质 === '温暖') { lower = lower - 5; upper = upper + 3; }
  return { lower: lower, upper: upper, range: lower + '~' + upper + '°C' };
}

/* 衣物保温值 (布衣+2 / 皮衣+5 / 防寒服+10) */
function _insulationValue(clothing) {
  if (!clothing) { return 0; }
  var total = 0;
  /* clothing可为字符串或数组 */
  if (typeof clothing === 'string') {
    var map = { '布衣': 2, '皮衣': 5, '防寒服': 10, '裸体': 0 };
    return map[clothing] || 0;
  }
  if (Array.isArray(clothing)) {
    var db = { '布衣': 2, '皮衣': 5, '防寒服': 10, '裸体': 0 };
    for (var i = 0; i < clothing.length; i++) {
      total += db[clothing[i]] || 0;
    }
  }
  return total;
}
