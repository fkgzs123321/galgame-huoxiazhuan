function _corpseState(hours) {
  var h = Number(hours) || 0;
  if (h < 6) { return '新鲜'; }

   公式以DDL骰子引擎为准,禁止本地修改 */
function _funeralEffect(type, attendees) {

  var att = Number(attendees) || 0;
  var actualAttendees = Math.min(att, base.参与人数);
  var ratio = base.参与人数 > 0 ? actualAttendees / base.参与人数 : 0;
  var moodAdj = Math.round(base.心情修正 * ratio * 100) / 100;

    类型: t,

  };

- 死亡后0-6小时: 新鲜 (处理难度×1.0, 全部产出可用)
- 死亡后6-24小时: 僵硬 (处理难度×1.2, 全部产出可用)
- 死亡后24-72小时: 腐败 (处理难度×1.5, 仅部分产出可用)
- 死亡后72-168小时: 严重腐败 (处理难度×2.0, 产出受限)
- 死亡后168小时以上: 白骨 (处理难度×0.8, 仅骨骼可用)
- 温度修正: 极寒×0.3 / 冰冻×0.6 / 常温×1.0 / 温暖×1.5 / 炎热×2.5

### 处理方式选择规则
- 土葬: 需墓地,基础处理方式,心情影响较小
- 火化: 需火化炉,卫生最佳,无疾病风险但心情影响较大
- 水葬: 需水域,耗时最短之一,但有中等疾病风险
- 天葬: 需露天高地,耗时最短,但疾病风险高
- 解剖研究: 需解剖台+医疗技能,产出研究样本,心情影响大
- 食用: 需屠宰台,产出肉×75,但心情影响极大(-40)且疾病风险极高

### 葬礼效果规则
- 心情修正 = 基础心情修正 × (实际参与人数 / 应到人数)
- 简朴葬礼: 3人参与,心情+5,银两消耗10
- 标准葬礼: 6人参与,心情+10,银两消耗30
- 隆重葬礼: 12人参与,心情+20,银两消耗80
- 无葬礼: 心情-15,社交影响-5,关系影响-3
- 银两不足时无法举办对应等级葬礼

### 解剖产出规则
- 医疗技能达标方可提取对应产出
- 产出率 = 基础产出率 / 尸体处理难度系数
- 新鲜尸体: 全部产出可用,产出率最高
- 严重腐败: 产出受限,产出率减半
- 白骨: 仅骨骼可用
- 器官需医疗技能≥12,大脑需医疗技能≥15

### 疾病风险规则
- 综合风险值 = min(100, (状态风险 + 处理风险) / 1.5)
- 风险等级: 无(0) / 低(<20) / 中(<50) / 高(<80) / 极高(≥80)
- 火化可消除所有疾病风险
- 食用方式疾病风险极高(90分)
- 严重腐败+任意处理方式风险均高

### 墓地建造规则
- 浅坑: 1小时建成,无估值,心情-5
- 标准墓: 2.5小时建成,估值50银,心情+2
- 石棺: 6小时建成,估值200银,心情+8
- 陵墓: 20小时建成,容量5,估值800银,心情+15
- 墓地容量满后需新建或选择其他处理方式

## 叙事要求

### 尸体处理叙事
- 描述尸体状态的视觉与嗅觉细节(颜色/气味/蝇虫)
- 体现殖民者对处理方式的心理反应(抵触/平静/悲伤)
- 解剖场景需体现医疗专业性与道德困境
- 食用场景需体现极端生存压力与心理创伤

### 葬礼叙事
- 体现葬礼氛围(简朴/标准/隆重的差异)
- 描述参与者情绪与社交互动
- 体现对死者的回忆与告别
- 银两消耗体现物质代价

### 墓地叙事
- 描述墓地建造过程与材料
- 体现墓地美观对殖民者心情的影响
- 陵墓作为高规格墓地需体现其庄严感

### 疾病风险叙事
- 高风险处理方式需体现卫生隐患
- 处理后需进行卫生检查
- 疾病爆发时需追溯处理过程
- 禁止数据化呈现风险数值,需转化为叙事描写
