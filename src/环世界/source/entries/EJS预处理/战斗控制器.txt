/* 多次命中判定: 冲锋枪3发/霰弹枪10弹丸/重机枪5发
   每发独立命中判定, 总伤害 = Σ各发伤害 */
function _multiHitCheck(skill, weaponName, baseDamage, seed) {
  var w = String(weaponName) || '';
  var shots = 1;
  if (w.indexOf('霰弹') >= 0) shots = 4;
  else if (w.indexOf('重机枪') >= 0) shots = 5;
  else if (w.indexOf('冲锋') >= 0) shots = 3;
  if (shots === 1) {
    return { 弹药数: 1, 命中数: 1, 总伤害: baseDamage, 命中详情: [{ 发数: 1, 命中: true, 伤害: baseDamage }] };
  }
  var totalDamage = 0;
  var hits = 0;
  var details = [];
  var baseHit = Math.min(95, 35 + (Number(skill) || 0) * 3);
  for (var i = 0; i < shots; i++) {
    var roll = _lcgRoll(seed + '_shot' + i);
    var hit = roll < baseHit;
    var dmg = hit ? baseDamage : 0;
    totalDamage += dmg;
    if (hit) hits++;
    details.push({ 发数: i + 1, 骰值: roll, 命中: hit, 伤害: dmg });
  }
  return { 弹药数: shots, 命中数: hits, 总伤害: Math.round(totalDamage * 100) / 100, 命中详情: details };
}
