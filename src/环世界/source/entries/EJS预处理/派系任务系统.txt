/* 任务触发判定
   faction=派系对象, relation=关系值, day=当前游戏日
   返回: 触发判定结果 */
function _questTrigger(faction, relation, day) {
  var rel = Number(relation) || 0;
  var d = Number(day) || 1;
  var lastDay = Number(getvar('stat_data.事件计时器.上次任务日', { scope: 'local', defaults: 0 })) || 0;
  var daysSince = d - lastDay;
  var minInterval;
  if (storyteller === '菲比') { minInterval = 5; }
  else if (storyteller === '兰迪') { minInterval = 2; }
  else { minInterval = 4; }
  var intervalMet = (daysSince >= minInterval);
  /* 敌对派系(-50以下)不触发任务 */
  if (rel <= -50) {
    return { 触发: false, 原因: '派系敌对,拒绝派发任务', 派系: faction.name, 关系: rel };
  }
  /* 中立以下派系仅触发求助信/赏金 */
  var availableTypes = [];
  if (rel < 10) {
    availableTypes = ['求助信', '赏金'];
  } else if (rel < 20) {
    availableTypes = ['求助信', '赏金', '物资请求'];
  } else {
    availableTypes = ['求助信', '赏金', '物资请求', '特殊任务'];
  }
  if (!intervalMet) {
    return {
      触发: false, 原因: '冷却中,距上次任务仅' + daysSince + '天',
      派系: faction.name, 关系: rel, 冷却剩余: minInterval - daysSince,
      可用类型: availableTypes
    };
  }
  /* LCG骰子判定是否触发(50%概率) */
  var triggerRoll = _lcgRoll('quest_trigger_' + faction.name + '_' + d, 1, 100)[0];
  var willTrigger = (triggerRoll <= 50);
  return {
    触发: willTrigger,
    原因: willTrigger ? '骰值' + triggerRoll + '/100 ≤50 触发' : '骰值' + triggerRoll + '/100 >50 未触发',
    派系: faction.name,
    关系: rel,
    距上次任务: daysSince + '天',
    可用类型: availableTypes,
    骰值: triggerRoll
  };
}

/* 难度计算
   quest=任务对象(查QUEST_DB), factionLevel=派系等级(派系实力1-5)
   返回: 难度星级与修正 */
function _questDifficulty(quest, factionLevel) {
  var q = quest;
  if (typeof quest === 'string') {
    q = QUEST_DB[quest] || QUEST_DB['求助信'];
  }
  var base = Number(q.难度基础) || 3;
  var fl = Number(factionLevel) || 1;
  /* 派系等级越高,任务难度越高 */
  var adjusted = Math.min(5, Math.max(1, base + Math.floor((fl - 1) / 2)));
  var diff = DIFFICULTY_DB[adjusted] || DIFFICULTY_DB[3];
  /* 阶段修正: 繁荣期任务难度+1 */
  if (phase === '繁荣期') { adjusted = Math.min(5, adjusted + 1); }
  diff = DIFFICULTY_DB[adjusted] || DIFFICULTY_DB[3];
  return {
    星级: diff.星级,
    名称: diff.名称,
    难度值: adjusted,
    检定修正: diff.检定修正,
    敌人强度: diff.敌人强度,
    基础难度: base,
    派系修正: Math.floor((fl - 1) / 2),
    阶段修正: (phase === '繁荣期') ? 1 : 0
  };
}

/* 奖励计算
   quest=任务对象, success=成功等级(大成功/普通成功/极难成功/失败/大失败)
   返回: 银两/关系/物品奖励 */
function _questReward(quest, success) {
  var q = quest;
  if (typeof quest === 'string') {
    q = QUEST_DB[quest] || QUEST_DB['求助信'];
  }
  var baseSilver = Number(q.奖励银两) || 100;
  var baseRelation = Number(q.奖励关系) || 5;
  var item = q.奖励物品 || '无';
  var silverReward = baseSilver;
  var relationReward = baseRelation;
  var itemReward = '无';
  var s = String(success) || '普通成功';
  if (s === '大成功') {
    silverReward = Math.round(baseSilver * 1.5);
    relationReward = baseRelation + 5;
    if (item !== '无') { itemReward = item + '(优质)'; }
    else { itemReward = '额外银两×0.5'; silverReward += Math.round(baseSilver * 0.5); }
  } else if (s === '极难成功' || s === '普通成功') {
    silverReward = baseSilver;
    relationReward = baseRelation;
    itemReward = item;
  } else if (s === '失败') {
    silverReward = 0;
    relationReward = -2;
    itemReward = '无';
  } else if (s === '大失败') {
    silverReward = 0;
    relationReward = -5;
    itemReward = '无';
  }
  /* 社交技能修正: 每点社交+2%银两 */
  var socialMod = 1 + p1Social * 0.02;
  silverReward = Math.round(silverReward * socialMod);
  return {
    银两: silverReward,
    关系: relationReward,
    物品: itemReward,
    成功等级: s,
    社交修正: socialMod,
    基础银两: baseSilver,
    基础关系: baseRelation
  };
}

/* 任务超时判定
   quest=任务对象, startDay=任务开始日
   返回: 超时判定结果(7天超时) */
function _questTimeout(quest, startDay) {
  var q = quest;
  if (typeof quest === 'string') {
    q = QUEST_DB[quest] || QUEST_DB['求助信'];
  }
  var timeoutDays = Number(q.超时天数) || 7;
  var start = Number(startDay) || 1;
  var elapsed = gameDay - start;
  var remaining = timeoutDays - elapsed;
  var isTimeout = (elapsed >= timeoutDays);
  return {
    超时: isTimeout,
    已用天数: elapsed,
    剩余天数: Math.max(0, remaining),
    超时限制: timeoutDays,
    失败原因: isTimeout ? '任务超时(' + timeoutDays + '天)' : '未超时'
  };
}
