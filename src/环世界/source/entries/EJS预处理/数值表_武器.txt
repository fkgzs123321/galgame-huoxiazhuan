function _combatMod(skill, weaponDamage, type) {
  var skl = Number(skill) || 0;
  var dmg = Number(weaponDamage) || 0;
  var hit;
  if (type === 'melee') {
    hit = Math.min(90, 40 + skl * 2.5);
  } else {
    hit = Math.min(95, 35 + skl * 3);
  }
  return {
    命中: hit,
    伤害: dmg * (0.5 + skl * 0.025),
    暴击率: Math.max(0, (skl - 10) * 2) / 100,
    瞄准轮数: Math.max(0.3, 1.2 - skl * 0.05)
  };
}
