function _growthStage(age){
  var a = Number(age) || 0;
  if(a <= 3) return '婴儿';
  if(a <= 6) return '幼儿';
  if(a <= 10) return '儿童';
  if(a <= 14) return '少年';
  if(a <= 18) return '青年';
  return '成年';
}

function _learnSkill(skill, education, age){
  var stage = _growthStage(age);
  var stageDb = GROWTH_STAGE_DB[stage] || GROWTH_STAGE_DB['婴儿'];
  var eduDb = EDUCATION_DB[education] || EDUCATION_DB['课堂教育'];
  var baseGain = 5 * stageDb.技能学习率 * eduDb.技能成长加成;
  var roll = _lcgRoll('learnSkill_' + skill + '_' + age, 1, 100)[0];
  var success = (roll <= eduDb.成功率 * 100);
  var gain = success ? Math.round(baseGain) : Math.round(baseGain * 0.3);
  return {
    skill: skill,
    stage: stage,
    education: education,
    success: success,
    gain: gain,
    learnRate: stageDb.技能学习率,
    bonus: eduDb.技能成长加成,
    detail: success ? '学习成功 +' + gain + '经验' : '学习失败 +' + gain + '经验'
  };
}

function _traitAwaken(age, mood){
  var stage = _growthStage(age);
  var stageDb = GROWTH_STAGE_DB[stage] || GROWTH_STAGE_DB['婴儿'];
  var awakenChance = stageDb.特质觉醒概率;
  var moodBonus = 1.0;
  if(mood > 80) moodBonus = 1.3;
  else if(mood > 60) moodBonus = 1.1;
  else if(mood < 30) moodBonus = 0.5;
  var finalChance = awakenChance * moodBonus;
  var roll = _lcgRoll('traitAwaken_' + age + '_' + mood, 1, 100)[0] / 100;
  if(roll > finalChance) return { triggered: false, trait: '无', detail: '未觉醒 (概率' + Math.round(finalChance * 100) + '%)' };
  var available = [];
  for(var k in TRAIT_AWAKEN_DB){
    if(TRAIT_AWAKEN_DB[k].觉醒年龄 <= age) available.push(k);
  }
  if(available.length === 0) return { triggered: false, trait: '无', detail: '无可用特质' };
  var idx = _lcgRoll('traitSelect_' + age, 0, available.length - 1)[0];
  var selected = available[idx];
  var traitDb = TRAIT_AWAKEN_DB[selected];
  return {
    triggered: true,
    trait: selected,
    effect: traitDb.效果,
    skillMod: traitDb.技能修正,
    moodEffect: traitDb.心情影响,
    ability: traitDb.特殊能力,
    detail: '觉醒特质[' + selected + ']: ' + traitDb.效果 + ' / ' + traitDb.特殊能力
  };
}

function _inheritSkill(parent1Skill, parent2Skill){
  var result = {};
  for(var i = 0; i < INHERIT_SKILL_LIST.length; i++){
    var sk = INHERIT_SKILL_LIST[i];
    var db = INHERIT_SKILL_DB[sk];
    var p1 = Number(getvar('stat_data.' + parent1Skill + '.' + sk, {scope:'local',defaults:5})) || 5;
    var p2 = Number(getvar('stat_data.' + parent2Skill + '.' + sk, {scope:'local',defaults:5})) || 5;
    var avg = (p1 + p2) / 2;
    var inherited = avg * db.继承率;
    var variance = _lcgRoll('inherit_' + sk + '_' + parent1Skill + '_' + parent2Skill, db.变异范围[0], db.变异范围[1])[0];
    var final = Math.round(inherited + variance);
    if(final > db.上限) final = db.上限;
    if(final < db.下限) final = db.下限;
    result[sk] = final;
  }
  return result;
}

function _comingOfAge(skills, traits){
  var total = 0;
  var count = 0;
  for(var k in skills){
    total += Number(skills[k]) || 0;
    count++;
  }
  var avg = count > 0 ? total / count : 0;
  var traitCount = traits.length;
  var ready = avg >= 5;
  var bonusTraits = traitCount >= 2;
  var finalBonus = 1.0;
  if(ready && bonusTraits) finalBonus = 1.3;
  else if(ready) finalBonus = 1.15;
  return {
    avgSkill: Number(avg.toFixed(1)),
    traitCount: traitCount,
    ready: ready,
    bonus: finalBonus,
    detail: '成年判定: 平均技能' + avg.toFixed(1) + ' / 特质' + traitCount + '个 / ' + (ready ? '达标' : '不达标') + ' / 成长系数×' + finalBonus
  };
}

## 教育规则
- **技能学习**：基础经验5 × 阶段学习率 × 教育加成，成功率决定全经验或30%经验
- **特质觉醒**：每阶段有觉醒概率，心情>80概率×1.3 / 心情>60概率×1.1 / 心情<30概率×0.5，仅可觉醒年龄≤当前年龄的特质
- **技能继承**：新生儿童年技能 = (父技能+母技能)/2 × 继承率 + 变异值，受上下限约束
- **童年影响**：心情修正 = 阶段心情修正 + 教育心情影响，疲劳 +教育耗时×2，受阶段食物需求和睡眠时间约束
- **教育心情**：艺术培养(+8)/社交培养(+5)为正向，劳动实践(-10)/战斗训练(-8)为负向
- **成年判定**：18岁且平均技能≥5为达标，达标+特质≥2个则成长系数×1.3，仅达标则×1.15
- **教育适用年龄**：课堂教育仅4-14岁可用，学徒制7岁起，劳动实践/战斗训练10-11岁起
