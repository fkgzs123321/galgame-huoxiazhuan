function _ruinExplore(researchSkill, ruinType, seed) {
  var ruin = RUIN_DB[ruinType] || RUIN_DB['古代仓库'];
  var threshold = ruin.解密难度 * 5;
  var check = _check(threshold, seed + '_ruin', '普通');
  var success = check.result === '普通成功' || check.result === '极难成功' || check.result === '大成功';
  var bonus = check.result === '大成功' ? 1.5 : 1.0;
  return {
    遗迹类型: ruinType,
    解密难度: ruin.解密难度,
    检定: check,
    解密成功: success,
    宝藏: success ? ruin.宝藏 : '部分(50%)',
    科技包: success ? Math.floor(ruin.科技包 * bonus) : 0,
    威胁: ruin.威胁
  };
}

function _ruinEncounter(threat, gameDay, seed) {
  if (threat <= 0) return { 遭遇: '无', 说明: '安全' };
  var roll = _lcgRoll(seed + '_encounter', 1, 100)[0];
  if (roll < 30) return { 遭遇: '古代机械体', 数量: threat, 说明: '守卫机械体苏醒' };
  if (roll < 50) return { 遭遇: '虫害', 数量: threat * 3, 说明: '巢穴虫类涌出' };
  if (roll < 70) return { 遭遇: '陷阱', 伤害: threat * 10, 说明: '自动防御系统触发' };
  if (roll < 85) return { 遭遇: '病原体', 强度: threat, 说明: '古代病毒释放' };
  return { 遭遇: '无', 说明: '虽然有威胁但未触发' };
}

function _ancientTechDisassemble(researchSkill, packCount, seed) {
  var skl = Number(researchSkill) || 0;
  var packs = Number(packCount) || 0;
  var researchPoints = 0;
  for (var i = 0; i < packs; i++) {
    var roll = _lcgRoll(seed + '_tech' + i, 1, 100)[0];
    var pts = Math.floor(skl * 10 * (0.5 + roll / 100));
    researchPoints += pts;
  }
  return { 科技包数: packs, 研究技能: skl, 获得研究点: researchPoints };
}
