/* _surgerySuccessRate(skill, difficulty, patientHealth) - 手术成功率计算 */
function _surgerySuccessRate(skill, difficulty, patientHealth) {
  var s = Number(skill) || 0;
  var d = Number(difficulty) || 1;
  var h = Number(patientHealth) || 80;
  /* 基础成功率: skill×4 + health×0.2 + 40 - difficulty×3 */
  var base = (s * 4) + (h * 0.2) + 40 - (d * 3);
  /* 难度惩罚: 难度>15时额外递减 */
  if (d > 15) { base -= (d - 15) * 4; }
  /* 年龄修正: 老年患者成功率下降 */
  if (pAge > 60) { base -= (pAge - 60) * 0.5; }
  /* 多次手术累积疲劳 */
  if (pSurgeryCount > 3) { base -= (pSurgeryCount - 3) * 2; }
  /* 上下限钳制 */
  if (base > 98) { base = 98; }
  if (base < 5) { base = 5; }
  return Math.round(base);
}

/* _anesthesiaRisk(patientHealth, duration) - 麻醉风险 */
function _anesthesiaRisk(patientHealth, duration) {
  var h = Number(patientHealth) || 80;
  var dur = Number(duration) || 1;
  /* 基础风险: (100-health)×0.3 + duration×1.5 */
  var base = ((100 - h) * 0.3) + (dur * 1.5);
  /* 健康低于40额外危险 */
  if (h < 40) { base += (40 - h) * 0.5; }
  /* 长时间麻醉(>6小时)陡增 */
  if (dur > 6) { base += (dur - 6) * 3; }
  if (base > 90) { base = 90; }
  if (base < 1) { base = 1; }
  var level;
  if (base < 10) { level = '极低'; }
  else if (base < 25) { level = '低'; }
  else if (base < 45) { level = '中'; }
  else if (base < 65) { level = '高'; }
  else { level = '极高'; }
  return {
    风险值: Math.round(base * 100) / 100,
    风险等级: level,
    麻醉时长: dur,
    健康基础: h,
    详情: '麻醉风险[' + level + '] 风险值' + Math.round(base * 100) / 100 + '% 时长' + dur + 'h'
  };
}

- **失败后果**：由手术类型决定(感染/大出血/器官衰竭/神经损伤/毁容/产妇风险/癌细胞扩散/畸形愈合/心脏骤停/脑损伤)
- **技能缓解**：失败后果死亡率/感染率 × 缓解系数(技能≥10为0.5/技能≥6为0.7/否则1.0)
- **器官排斥**：前7天急性期×1.5 / 8-14天亚急性×1.0 / 15-30天×0.6 / 30天后慢性×0.3
- **麻醉风险**：(100-health)×0.3 + duration×1.5；健康<40额外惩罚；麻醉>6h陡增
- **术后恢复**：每日恢复 = 5 × careFactor + skill×0.3；护理等级(无护理0.6/基础1.0/高级1.2/专业1.5)
- **血型匹配**：心脏/肺必须匹配，肝/肾/胃建议匹配，皮肤不要求
- **供体时效**：心脏≤6h / 肺≤8h / 胃≤10h，超时存活率减半

## 叙事要求
- 难度≥17(极高)的手术(心脏手术/脑部手术)叙事须描写手术室紧张氛围、生命体征波动、汗湿的手套
- 手术失败时叙事须依据失败后果类型描写(感染化脓/血液喷涌/器官颜色异常/神经痉挛)
- 器官移植急性排斥期(前7天)叙事须描写患者高热、移植部位肿痛、免疫抑制剂的副作用
- 仿生体植入后叙事须描写神经接口接入的刺痛、机械部件与肉体的磨合、不协调的抽动
- 麻醉风险"极高"时叙事须描写患者瞳孔变化、呼吸抑制、监护仪警报
- 术后恢复"危重期"叙事须描写患者虚弱、伤口渗液、间歇性意识模糊
- 截肢手术叙事须描写骨锯的震动声、断面的处理、患者醒来后的幻肢痛
- 剖腹产手术叙事须描写新生儿的哭声、产妇的疲惫、保温箱的暖光
