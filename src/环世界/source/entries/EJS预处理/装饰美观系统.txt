function _aestheticScore(buildings, environmentFactors){
  var total = 50;
  for(var i = 0; i < buildings.length; i++){
    var db = AESTHETIC_BUILDING_DB[buildings[i].type] || AESTHETIC_BUILDING_DB['木墙'];
    total += db.美观 * (buildings[i].count || 1);
  }
  for(var j = 0; j < environmentFactors.length; j++){
    var ef = ENVIRONMENT_FACTOR_DB[environmentFactors[j].type] || ENVIRONMENT_FACTOR_DB['整洁地面'];
    total += ef.美观;
  }
  return Math.max(0, Math.min(100, total));
}

function _aestheticMood(score){
  for(var k in AESTHETIC_MOOD_DB){

}
