/* 获取殖民者工作优先级
   colonist=殖民者对象, workType=工作类型(查WORK_TYPE_DB)
   返回: 0-4 优先级数字,0表示禁用 */
function _workPriority(colonist, workType) {
  var wt = String(workType) || '休闲';
  var workDef = WORK_TYPE_DB[wt];
  if (!workDef) { return 0; }
  var basePriority = Number(workDef.优先级) || 4;
  var mood = Number(colonist.心情) || 50;
  var fatigue = Number(colonist.疲劳) || 0;
  /* 心情<20紧急工作优先级降低(消极怠工) */
  if (mood < 20 && workDef.类别 !== '紧急') { return 0; }
  /* 疲劳>=80仅保留紧急工作 */
  if (fatigue >= 80 && workDef.类别 !== '紧急') { return 0; }
  /* 警戒等级>=2时紧急工作强制为1 */
  var alertLvl = Number(getvar('stat_data.殖民地.警戒等级', { scope: 'local', defaults: 20 })) || 20;
  if (alertLvl >= 2 && workDef.类别 === '紧急') { return 1; }
  return basePriority;
}

/* 分配工作给殖民者
   colonist=殖民者对象, workType=工作类型, priority=1-4优先级
   返回: 分配结果对象 */
function _workAssign(colonist, workType, priority) {
  var wt = String(workType) || '休闲';
  var workDef = WORK_TYPE_DB[wt];
  if (!workDef) {
    return { 成功: false, 原因: '未知工作类型: ' + wt, 殖民者: colonist.姓名, 工作: wt, 优先级: 0 };
  }
  var p = Number(priority);
  if (p < 0 || p > 4) { p = workDef.优先级; }
  if (p === 0) {
    return { 成功: true, 原因: '已禁用该工作', 殖民者: colonist.姓名, 工作: wt, 优先级: 0 };
  }
  var mood = Number(colonist.心情) || 50;
  var fatigue = Number(colonist.疲劳) || 0;
  /* 技能匹配检查 */
  var requiredSkill = workDef.技能;
  var skillVal = 0;
  if (requiredSkill !== '无') {
    skillVal = Number(colonist[requiredSkill]) || 0;
  }
  /* 效率修正 */
  var moodMod = 1.0;
  if (mood < 30) { moodMod = 0.8; }
  else if (mood > 70) { moodMod = 1.1; }
  var fatigueMod = 1.0;
  if (fatigue > 80) { fatigueMod = 0.5; }
  else if (fatigue > 60) { fatigueMod = 0.7; }
  var efficiency = Math.round(moodMod * fatigueMod * 100) / 100;
  return {
    成功: true,
    原因: '分配成功',
    殖民者: colonist.姓名,
    工作: wt,
    类别: workDef.类别,
    优先级: p,
    所需技能: requiredSkill,
    技能值: skillVal,
    心情修正: moodMod,
    疲劳修正: fatigueMod,
    综合效率: efficiency
  };
}

/* 时段工作调度
   time=时段(清晨/上午/中午/下午/傍晚/夜晚/深夜), priorities=优先级映射
   返回: 该时段推荐工作分配表 */
function _workSchedule(time, priorities) {
  var t = String(time) || '清晨';
  var sched = SCHEDULE_DB[t];
  if (!sched) { sched = SCHEDULE_DB['清晨']; }
  var recommended = sched.推荐工作;
  var result = [];
  for (var i = 0; i < recommended.length; i++) {
    var wt = recommended[i];
    var workDef = WORK_TYPE_DB[wt];
    if (!workDef) { continue; }
    var prio = workDef.优先级;
    /* 如果传入了自定义优先级映射,覆盖默认优先级 */
    if (priorities && priorities[wt] != null) {
      prio = Number(priorities[wt]) || workDef.优先级;
    }
    result.push({
      工作: wt,
      类别: workDef.类别,
      优先级: prio,
      所需技能: workDef.技能,
      备注: workDef.备注
    });
  }
  /* 按优先级排序(1最高) */
  result.sort(function(a, b) { return a.优先级 - b.优先级; });
  return { 时段: t, 说明: sched.备注, 推荐工作: result };
}
