function _dailySocial(colonistCount, seed){
  if(colonistCount < 2) return [];

  var events = [];
  for(var i = 0; i < pairs; i++){
    var roll = _lcgRoll((seed||'social') + '_' + i, 1, 100)[0] / 100;
    var cum = 0;
    for(var k in SOCIAL_EVENT_DB){
      cum += SOCIAL_EVENT_DB[k].触发;
      if(roll < cum){

}
