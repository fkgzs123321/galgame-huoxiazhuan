/* 轨道贸易触发判定
   signal=是否有贸易信号器(0/1), day=当前游戏日
   返回: 触发判定结果 */
function _orbitalTradeTrigger(signal, day) {
  var sig = Number(signal) || 0;
  var d = Number(day) || 1;
  if (sig !== 1) {
    return { 触发: false, 原因: '未建造贸易信号器,无法呼叫轨道贸易', 需建造: '贸易信号器(组件×5+钢铁×10)' };
  }
  var daysSince = d - lastOrbitalDay;
  /* 检查各商船类型是否到达来访间隔 */
  var availableTraders = [];
  var traderNames = Object.keys(TRADER_DB);
  for (var i = 0; i < traderNames.length; i++) {
    var tName = traderNames[i];
    var trader = TRADER_DB[tName];
    var interval = Number(trader.间隔) || 7;
    if (daysSince >= interval) {
      availableTraders.push({
        类型: tName,
        间隔: interval,
        商品: trader.商品,
        距上次: daysSince + '天'
      });
    }
  }
  if (availableTraders.length === 0) {
    var nextMinInterval = 7;
    var nextTrader = '补给船';
    for (var j = 0; j < traderNames.length; j++) {
      var tn = traderNames[j];
      var intv = Number(TRADER_DB[tn].间隔) || 7;
      var remain = intv - daysSince;
      if (remain > 0 && remain < nextMinInterval) {
        nextMinInterval = remain;
        nextTrader = tn;
      }
    }
    return {
      触发: false,
      原因: '无商船到达,距下次最近商船(' + nextTrader + ')还有' + nextMinInterval + '天',
      距上次贸易: daysSince + '天',
      下次商船: nextTrader,
      剩余天数: nextMinInterval
    };
  }
  /* LCG骰子选择来访商船(从可用商船中随机选一个) */
  var selectRoll = _lcgRoll('orbital_select_' + d, 1, availableTraders.length)[0];
  var selected = availableTraders[selectRoll - 1];
  return {
    触发: true,
    原因: '商船到达,可进行轨道贸易',
    商船类型: selected.类型,
    商品范围: selected.商品,
    距上次贸易: selected.距上次,
    可用商船: availableTraders,
    选中商船: selected.类型,
    骰值: selectRoll
  };
}

/* 轨道贸易买入价(比地面贸易贵10%)
   item=商品基础估值, traderType=商船类型(查TRADER_DB)
   返回: 买入价 */
function _orbitalBuyPrice(item, traderType) {
  var baseValue = Number(item) || 1;
  var tt = String(traderType) || '补给船';
  var trader = TRADER_DB[tt] || TRADER_DB['补给船'];
  var buyMod = Number(trader.买入修正) || 1.1;
  /* 社交技能修正: 每点社交-1%买入价 */
  var socialMod = Math.max(0.5, 1 - p1Social * 0.01);
  var raw = baseValue * buyMod * socialMod;
  return Math.round(raw * 100) / 100;
}

/* 轨道贸易卖出价(比地面贸易高15%)
   item=商品基础估值, traderType=商船类型(查TRADER_DB)
   返回: 卖出价 */
function _orbitalSellPrice(item, traderType) {
  var baseValue = Number(item) || 1;
  var tt = String(traderType) || '补给船';
  var trader = TRADER_DB[tt] || TRADER_DB['补给船'];
  var sellMod = Number(trader.卖出修正) || 1.15;
  /* 社交技能修正: 每点社交+1%卖出价 */
  var socialMod = 1 + p1Social * 0.01;
  var raw = baseValue * sellMod * socialMod;
  return Math.round(raw * 100) / 100;
}

/* 轨道贸易成功率
   skill=社交技能, amount=交易银两总额
   返回: 贸易检定结果 */
function _orbitalTradeSuccess(skill, amount) {
  var skl = Number(skill) || 0;
  var amt = Number(amount) || 0;
  var threshold = skl * 5;
  /* 大额交易难度提升: 每100银两阈值-5 */
  var amountPenalty = Math.floor(amt / 100) * 5;
  var adjustedThreshold = Math.max(5, threshold - amountPenalty);
  var checkResult = _check(adjustedThreshold, 'orbital_trade_' + gameDay, '普通');
  return {
    阈值: adjustedThreshold,
    基础阈值: threshold,
    大额惩罚: amountPenalty,
    交易额: amt,
    骰值: checkResult.roll,
    结果: checkResult.result,
    成功: (checkResult.result !== '失败' && checkResult.result !== '大失败')
  };
}
