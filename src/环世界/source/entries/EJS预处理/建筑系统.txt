/* 建造耗时公式: 实际时间 = 基础耗时 / (1 + 建设技能 x 0.08) */
function _buildTime(baseTime, skill) {
  var skl = Number(skill) || 0;
  var actual = baseTime / (1 + skl * 0.08);
  return Math.round(actual * 100) / 100;
}

  if (skl <= 3) {
    return { grade: '糟糕', hpMod: 0.8, note: '糟糕品质' };

}
