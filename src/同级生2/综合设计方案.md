### 3.2 关键表设计骨架（P0 必须实现）

#### 3.2.1 `global_state`（单行表）

```sql
CREATE TABLE IF NOT EXISTS global_state (
  row_id INTEGER PRIMARY KEY CHECK(row_id = 1),
  -- 时间（6 列，对应 MVU schema.ts 的 6 个时间变量）
  day_count INTEGER NOT NULL CHECK(day_count BETWEEN 1 AND 17),
  current_date TEXT NOT NULL CHECK(LENGTH(current_date) = 5),  -- 'MM-DD'
  weekday TEXT NOT NULL CHECK(weekday IN ('周一','周二','周三','周四','周五','周六','周日')),
  time_slot TEXT NOT NULL CHECK(time_slot IN ('早','上午','下午','晚','深夜')),
  time_slot_idx INTEGER NOT NULL CHECK(time_slot_idx BETWEEN 0 AND 4),
  chapter_idx INTEGER NOT NULL CHECK(chapter_idx BETWEEN 1 AND 4),  -- 1=前奏 2=核心 3=尾声 4=结局
  -- 场景（3 列）
  current_region TEXT NOT NULL CHECK(current_region IN ('自宅周边','八十八学园','八十八町商业区','八十八海岸','温泉乡','88市民医院','保育园')),
  current_location TEXT NOT NULL,
  scene_mode TEXT NOT NULL CHECK(scene_mode IN ('上学','事件','休息','打工','约会')),
  -- 难度（1 列）
  difficulty TEXT NOT NULL CHECK(difficulty IN ('轻松','普通','挑战','地狱')) DEFAULT '普通',
  -- 天气（1 列）
  weather TEXT CHECK(weather IN ('晴','阴','雨','雪','风')),
  -- 当前女角（2 列）
  current_heroine_id INTEGER NOT NULL CHECK(current_heroine_id BETWEEN 0 AND 19) DEFAULT 0,
  current_heroine_name TEXT,
  -- 不可逆状态计数（4 列）
  illegal_count INTEGER NOT NULL CHECK(illegal_count >= 0) DEFAULT 0,
  bad_end_triggered INTEGER NOT NULL CHECK(bad_end_triggered IN (0,1)) DEFAULT 0,
  bad_end_type TEXT,
  -- 时间戳（1 列）
  last_updated TEXT NOT NULL
);
```
