/* 读取脚本变量中的配置, 合并到 CFG; 缺项用默认值兜底 */
function loadConfig() {
  try {
    const vars = getVariables({ type: 'script' });
    const saved = vars && vars[CONFIG_VAR_KEY];
    if (saved && typeof saved === 'object') {
      CFG = {
        bookName  : (saved.bookName   || DEFAULTS.bookName),
        entryName : (saved.entryName  || DEFAULTS.entryName),
        presetName: (saved.presetName || DEFAULTS.presetName),
      };
    }
  } catch (_) { /* 读取失败则保持默认值 */ }
}

/* 把当前 CFG 写回脚本变量(持久化) */
async function saveConfig() {
  try {
    await insertOrAssignVariables({ [CONFIG_VAR_KEY]: { ...CFG } }, { type: 'script' });
    return true;
  } catch (e) {
    notify('保存配置失败: ' + (e && e.message), 'err');
    return false;
  }
}

  topDocument.getElementById(`${PANEL_ID}-config`)?.remove();
  try { eventClearEvent(getButtonEvent(BTN_NAME)); } catch (_) {}
}
disposePrevious();

/* ========= helpers ========= */
function getApi() {
  return window.AutoCardUpdaterAPI || (window.parent && window.parent.AutoCardUpdaterAPI) || null;
}
function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }
async function waitApi(retry = API_RETRY, interval = API_INTERVAL) {
  for (let i = 0; i < retry; i++) {
    const api = getApi();
    if (api) return api;
    await sleep(interval);
  }
  return null;
}
function requireMethod(api, name) {
  if (!api || typeof api[name] !== 'function') {
    throw new Error(`plugin missing method ${name}() — version mismatch or not ready`);
  }
  return api[name].bind(api);
}
function escapeHtml(s) {
  return String(s == null ? '' : s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function notify(text, kind) {
  try {
    if (typeof toastr !== 'undefined') {
      (toastr[kind === 'ok' ? 'success' : kind === 'err' ? 'error' : 'info'])(text, '同级生2模板');
    }
  } catch (_) {}
  const el = topDocument.getElementById(`${PANEL_ID}-status`);
  if (el) el.textContent = text;
}

/* ============================================================
 * FIX CORE: visualViewport 实时居中浮层, 抵消软键盘顶屏

const vv = topWindow.visualViewport || null;

function placeOne(el) {
  if (!el) return;
  let w, h, ox, oy;
  if (vv) { w = vv.width; h = vv.height; ox = vv.offsetLeft; oy = vv.offsetTop; }
  else    { w = topWindow.innerWidth; h = topWindow.innerHeight; ox = 0; oy = 0; }
  el.style.maxHeight = Math.max(160, Math.floor(h - 24)) + 'px';
  el.style.left = (ox + w / 2) + 'px';
  el.style.top  = (oy + h / 2) + 'px';
  el.style.transform = 'translate(-50%,-50%)';
}
function reposition() {
  if (panelEl && panelEl.classList.contains('is-open')) placeOne(panelEl);
  if (confirmEl && confirmEl.classList.contains('is-open')) {
    const box = confirmEl.querySelector('.cf-box'); if (box) placeOne(box);
  }
  if (configEl && configEl.classList.contains('is-open')) {
    const box = configEl.querySelector('.cf-box'); if (box) placeOne(box);
  }
}
let rafPending = false;
function onViewportChange() {
  if (rafPending) return;
  rafPending = true;
  topWindow.requestAnimationFrame(() => { rafPending = false; reposition(); });
}
if (vv) {
  vv.addEventListener('resize', onViewportChange, SIG);
  vv.addEventListener('scroll', onViewportChange, SIG);
} else {
  topWindow.addEventListener('resize', onViewportChange, SIG);
}
/* 给浮层内输入框统一挂 focus/blur 复位 */
function bindInputReposition(scope) {
  scope.querySelectorAll('input').forEach(inp => {
    inp.addEventListener('focus', () => setTimeout(reposition, 300), SIG);
    inp.addEventListener('blur',  () => setTimeout(reposition, 300), SIG);
  });
}

  overlayEl.classList.add('is-open');

  overlayEl?.classList.remove('is-open');
  confirmEl?.classList.remove('is-open');
  configEl?.classList.remove('is-open');

}

/* ========= refresh preset list ========= */
async function refreshPresetList() {
  const box = topDocument.getElementById(`${PANEL_ID}-list`);
  try {
    const api = getApi();
    if (!api) { if (box) box.textContent = '插件未就绪'; return; }
    const names = await requireMethod(api, 'getTemplatePresetNames')();
    if (box) box.textContent = (Array.isArray(names) && names.length) ? names.join('\n') : '(暂无预设)';
  } catch (e) {
    if (box) box.textContent = '读取失败';
    notify('刷新预设失败: ' + (e && e.message), 'err');
  }
}

/* ========= refresh worldbook ========= */
async function refreshWorldbook() {
  try {
    const api = await waitApi();
    if (!api) throw new Error('插件未就绪');
    if (typeof api.refreshDataAndWorldbook === 'function') await api.refreshDataAndWorldbook();
    else if (typeof api.syncWorldBookEntries === 'function') await api.syncWorldBookEntries({ createIfNeeded: true });
    else throw new Error('缺少 refreshDataAndWorldbook / syncWorldBookEntries 接口');
    notify('世界书已刷新并重新注入', 'ok');
  } catch (e) {
    notify('刷新世界书失败: ' + (e && e.message), 'err');
  }
}

/* ========= read WI entry -> extract single-line JSON ========= */
async function readTemplateObject() {
  let entries;
  try {
    entries = await getWorldbook(CFG.bookName);
  } catch (e) {
    throw new Error(`read worldbook "${CFG.bookName}" failed: ${e && e.message}`);
  }
  if (!Array.isArray(entries) || !entries.length) {
    throw new Error(`worldbook "${CFG.bookName}" empty or missing`);
  }
  const hit = entries.find(en => (en.comment || en.name || '') === CFG.entryName);
  if (!hit) {
    throw new Error(`entry "${CFG.entryName}" not found (it should stay DISABLED but still readable)`);
  }
  let raw = String(hit.content == null ? '' : hit.content).replace(/^\uFEFF/, '').trim();
  if (raw[0] !== '{' && raw[0] !== '[') {
    const s = raw.indexOf('{'), e = raw.lastIndexOf('}');
    if (s !== -1 && e !== -1 && e > s) raw = raw.slice(s, e + 1);
  }
  try {
    return JSON.parse(raw);
  } catch (e) {
    throw new Error(`template JSON parse failed (single-line compact JSON? no {{macro}}?): ${e && e.message}`);
  }
}

/* ========= core: read entry JSON -> create template -> conflict -> verify ========= */
let busy = false;
async function createTemplate() {
  if (busy) return;
  busy = true;
  try {
    notify('读取条目中…', 'info');
    const api = await waitApi();
    if (!api) throw new Error('插件未就绪(等待超时)');
    const importTpl = requireMethod(api, 'importTemplateFromData');
    const listTpl   = requireMethod(api, 'getTemplatePresetNames');

    const tplData = await readTemplateObject();

    let presetName = CFG.presetName;   // 预设名取自配置

    const existing = await listTpl();
    if (Array.isArray(existing) && existing.includes(presetName)) {
      const r = await askOnConflict(presetName);
      if (r.action === 'cancel') { notify('已取消创建', 'info'); return; }
      if (r.action === 'rename') presetName = r.name;
    }

    notify('创建模板中…', 'info');
    const res = await importTpl(tplData, { scope: scopeState, presetName });
    if (res && res.success === false) throw new Error(res.message || '插件返回 success:false');

    const verify = await listTpl();
    const ok = Array.isArray(verify) && verify.includes(presetName);
    if (ok && typeof api.switchTemplatePreset === 'function') {
      try { await api.switchTemplatePreset(presetName, { scope: scopeState }); } catch (_) {}
    }
    await refreshPresetList();
    notify(ok ? `已创建并切换: ${presetName} (${scopeState})` : `已提交但未校验到, 请点"刷新预设列表"复查`, ok ? 'ok' : 'err');
  } catch (e) {
    notify('创建失败: ' + (e && e.message), 'err');
  } finally {
    busy = false;
  }
}
