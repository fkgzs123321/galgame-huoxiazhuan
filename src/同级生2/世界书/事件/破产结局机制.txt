// 六、破产预警检测函数（每日 0 点 + 每时段检测）
// ────────────────────────────────────────────────────────────
const BANKRUPTCY_CHECK = {
  // 每日 0 点检测（D0系统控制器调用）
  daily_check: function() {
    const _money = Number(getvar('stat_data.主角.现金', { scope: 'local', defaults: 5000 })) || 0;
    const _consecutiveNoIncome = Number(getvar('stat_data.临时.连续无收入天数', { scope: 'local', defaults: 0 })) || 0;
    const _hunger = Number(getvar('stat_data.主角.饥饿', { scope: 'local', defaults: 30 })) || 0;
    const _mood = Number(getvar('stat_data.主角.心情', { scope: 'local', defaults: 70 })) || 0;
    const _difficulty = getvar('stat_data.临时.当前难度', { scope: 'local', defaults: '普通' });
    
    // 1. 判断当前破产阶段
    let _currentStage = 'stage_0_safe';
    if (_money <= 0 && _consecutiveNoIncome >= 1) {
      _currentStage = 'stage_5_emergency';
    } else if (_money <= 0) {
      _currentStage = 'stage_4_critical';
    } else if (_money <= 500) {
      _currentStage = 'stage_3_danger';
    } else if (_money <= 2000) {
      _currentStage = 'stage_2_warning';
    } else if (_money <= 5000) {
      _currentStage = 'stage_1_caution';
    }
    
    // 2. 检测是否触发破产结局
    const _diffMod = BANKRUPTCY_DIFFICULTY_MODIFIERS[_difficulty] || BANKRUPTCY_DIFFICULTY_MODIFIERS['普通'];
    const _threshold = _diffMod.consecutive_days_threshold;
    
    if (_money <= 0 && _consecutiveNoIncome >= _threshold) {
      // 检测3种变体
      if (_hunger >= 100) {
        return { triggered: true, ending: 'bankruptcy_starvation', stage: _currentStage };
      }
      if (_mood <= 0) {
        return { triggered: true, ending: 'bankruptcy_mental_collapse', stage: _currentStage };
      }
      if (_hunger >= 90) {
        return { triggered: true, ending: 'bankruptcy_standard', stage: _currentStage };
      }
    }
    
    // 3. 检测债务陷阱
    const _westTempleDebt = getvar('stat_data.隐藏.西寺债务陷阱', { scope: 'local', defaults: 0 });
    const _dayCount = Number(getvar('stat_data.时间.天数', { scope: 'local', defaults: 1 })) || 1;
    if (_westTempleDebt === 1 && _dayCount >= 15 && _money < 15000) {
      return { triggered: true, ending: 'bankruptcy_debt', stage: _currentStage };
    }
    
    return { triggered: false, stage: _currentStage };
  },
  
  // 每时段检测（移动/行动后调用）
  slot_check: function() {
    const _money = Number(getvar('stat_data.主角.现金', { scope: 'local', defaults: 5000 })) || 0;
    const _stage = this.daily_check().stage;
    
    // stage_4~5 时限制社交行动
    if (_stage === 'stage_4_critical' || _stage === 'stage_5_emergency') {
      return {
        social_actions_locked: true,
        forced_actions: ['work', 'sell_items', 'borrow'],
        alert_message: BANKRUPTCY_WARNING_STAGES[_stage].alert_message
      };
    }
    
    return { social_actions_locked: false };
  }
};
