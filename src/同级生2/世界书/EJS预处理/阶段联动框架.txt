  const v = Number(aff) || 0;
  const _specialStages = ['攻略完成', '失恋'];
  for (const key in STAGE_FRAMEWORK) {
    if (_specialStages.indexOf(key) !== -1) continue;  // 跳过特殊态
    const s = STAGE_FRAMEWORK[key];
    if (v >= s.range[0] && v <= s.range[1]) return s;
  }

/* ─── 辅助函数：根据阶段名获取框架（支持特殊态） ─── */
this.getStageByName = function getStageByName(stageName) {
  if (stageName === '攻略完成') return STAGE_FRAMEWORK['攻略完成'];
  if (stageName === '失恋') return STAGE_FRAMEWORK['失恋'];
  return STAGE_FRAMEWORK[stageName] || STAGE_FRAMEWORK['初识'];
}

/* ─── 辅助函数：检查物品是否被当前阶段接受 ─── */
this.checkItemAcceptance = function checkItemAcceptance(stageName, itemCategory) {
  const s = STAGE_FRAMEWORK[stageName];
  if (!s) return { accepted: false, reason: '阶段不存在' };
  if (s.items.accept.includes('全部')) return { accepted: true, reason: '亲密阶段全部接受' };
  if (s.items.accept.includes(itemCategory)) return { accepted: true, reason: '当前阶段接受' };
  if (s.items.refuse.includes(itemCategory)) return { accepted: false, reason: '当前阶段拒绝' };
  if (s.items.hate.includes(itemCategory)) return { accepted: false, reason: '当前阶段讨厌', reverse_effect: true };

this.checkNsfwStageAllowed = function checkNsfwStageAllowed(stageName, nsfwAction) {
  const s = STAGE_FRAMEWORK[stageName];
  if (!s) return { allowed: false, reason: '阶段不存在' };
