    if (_drinkScore >= 80) { _tier = '大成功'; _effectMult = 1.5; _changes.thirst = -35; _changes.mood = 3; _desc = '清凉解渴，神清气爽'; }
    else if (_drinkScore >= 50) { _tier = '成功'; _effectMult = 1.0; _changes.thirst = -25; _desc = '解渴'; }
    else { _tier = '普通'; _effectMult = 0.5; _changes.thirst = -15; _desc = '稍微润喉'; }
  }
  else if (_act === 'bath') {
    /* 洗澡判定：清洁度越低效果越好 */
    var _bathMod = _hygiene <= 20 ? 20 : _hygiene <= 40 ? 10 : 0;
    var _bathScore = _dice + _bathMod;
    if (_bathScore >= 80) { _tier = '大成功'; _effectMult = 1.5; _changes.hygiene = 50; _changes.mood = 10; _changes.fatigue = -5; _desc = '舒爽的沐浴，焕然一新'; }

    else { _tier = '普通'; _effectMult = 0.5; _changes.hygiene = 20; _desc = '简单冲洗'; }
  }
  else if (_act === 'rest') {
    /* 休息/睡眠判定：疲劳越高效果越好，但心情过低影响睡眠质量 */
    var _restMod = _fatigue >= 60 ? 20 : _fatigue >= 30 ? 10 : 0;
    _restMod += _mood <= 20 ? -15 : 0;
    var _restScore = _dice + _restMod;
    if (_restScore >= 80) { _tier = '大成功'; _effectMult = 1.5; _changes.fatigue = -40; _changes.physical = 30; _changes.mood = 5; _desc = '深度睡眠，精力充沛'; }
    else if (_restScore >= 50) { _tier = '成功'; _effectMult = 1.0; _changes.fatigue = -25; _changes.physical = 20; _desc = '休息良好，恢复体力'; }
    else if (_restScore >= 25) { _tier = '普通'; _effectMult = 0.5; _changes.fatigue = -15; _changes.physical = 10; _desc = '浅眠，略微恢复'; }
    else { _tier = '失败'; _effectMult = 0; _changes.fatigue = -5; _changes.mood = -3; _desc = '失眠/噩梦，越睡越累'; }
  }
  else if (_act === 'shop') {
    /* 购物判定：声誉修正购物体验 */
    var _shopMod = _willpower >= 60 ? 10 : _willpower <= 20 ? -10 : 0;
    var _shopScore = _dice + _shopMod;
    if (_shopScore >= 80) { _tier = '大成功'; _effectMult = 1.5; _changes.mood = 6; _desc = '买到心仪好物，心情大好'; }
    else if (_shopScore >= 50) { _tier = '成功'; _effectMult = 1.0; _changes.mood = 3; _desc = '顺利购物'; }
    else if (_shopScore >= 25) { _tier = '普通'; _effectMult = 0.5; _changes.mood = 1; _desc = '普通购物'; }
    else { _tier = '失败'; _effectMult = 0; _changes.mood = -3; _desc = '没买到想要的东西/被坑'; }
  }
  else if (_act === 'commute') {
    /* 通勤/移动判定：体力影响 */
    var _commuteMod = _physical >= 60 ? 10 : _physical <= 20 ? -15 : 0;
    var _commuteScore = _dice + _commuteMod;
    if (_commuteScore >= 70) { _tier = '大成功'; _effectMult = 1.5; _changes.fatigue = 1; _desc = '一路畅通，心情愉快'; }
    else if (_commuteScore >= 40) { _tier = '成功'; _effectMult = 1.0; _changes.fatigue = 2; _desc = '顺利到达'; }
    else { _tier = '普通'; _effectMult = 0.5; _changes.fatigue = 4; _changes.mood = -1; _desc = '路上有些波折'; }
  }
  else if (_act === 'save_money') {
    /* 存钱判定：无骰子风险，固定成功（仅记录） */
    _tier = '成功'; _effectMult = 1.0; _desc = '储蓄存入银行';
  }
  else {
    /* 未知日常动作：兜底普通 */
    _tier = '普通'; _effectMult = 0.5; _desc = '日常行为';
  }

  return {
    action: _act,
    tier: _tier,
    effect_mult: _effectMult,
    dice_value: Math.round(_dice * 100) / 100,
    changes: _changes,
    desc: _desc
  };
}
