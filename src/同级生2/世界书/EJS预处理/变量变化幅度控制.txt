/* ───────── 六、变化校验函数（思维链第 6 步强制调用） ─────────
   参数：
     variable_type    - 变量分类（CHANGE_LIMITS 的 key）
     variable_name    - 具体变量名（如 'affection'）
     current_value    - 当前值
     change_amount    - 拟变化量（正/负）
     difficulty       - 难度（'轻松'/'普通'/'挑战'/'地狱'）
   返回：
     { valid, adjusted, clamped, reason }
     - valid=true 时 adjusted 为难度修正后的最终变化量
     - valid=false 时 clamped 为截断后的变化量，reason 为警告 */
function validateChange(variable_type, variable_name, current_value, change_amount, difficulty) {
  const _limits = CHANGE_LIMITS[variable_type];
  if (!_limits || !_limits[variable_name]) {
    return { valid: true, adjusted: change_amount, reason: '无限制（变量未注册）' };
  }
  const _limit = _limits[variable_name];

  // 应用难度修正（按变量类型 + 变化方向）
  const _diffMod = DIFFICULTY_CHANGE_MODIFIERS[difficulty] || DIFFICULTY_CHANGE_MODIFIERS['普通'];
  let _adjustedChange = change_amount;

  if (change_amount > 0) {
    // 正向变化（增长）
    if (variable_name === 'affection') {
      _adjustedChange = change_amount * _diffMod.affection_gain_mult;
    } else if (variable_name === 'money' && variable_type === 'economy') {
      _adjustedChange = change_amount * _diffMod.money_gain_mult;
    } else if (variable_type === 'skills') {
      _adjustedChange = change_amount * _diffMod.skill_gain_mult;
    } else if (variable_name === 'jealousy') {
      _adjustedChange = change_amount * _diffMod.jealousy_gain_mult;
    }
  } else if (change_amount < 0) {
    // 负向变化（下降）
    if (variable_name === 'affection') {
      _adjustedChange = change_amount * _diffMod.affection_loss_mult;
    }
    // 其他变量负向变化暂不应用难度修正，由 全局表.decrease_mult 统一处理
  }

  // 四舍五入到整数（避免浮点误差）
  _adjustedChange = Math.round(_adjustedChange);

  // 校验单次上限（截断而非拒绝，保证剧情流畅）
  if (_adjustedChange > _limit.single_change_max) {
    return {
      valid: false,
      clamped: _limit.single_change_max,
      reason: `单次变化超限：${variable_name} 拟变化 ${_adjustedChange} > 上限 ${_limit.single_change_max}，已截断`
    };
  }
  if (_adjustedChange < _limit.single_change_min) {
    return {
      valid: false,
      clamped: _limit.single_change_min,
      reason: `单次变化超限：${variable_name} 拟变化 ${_adjustedChange} < 下限 ${_limit.single_change_min}，已截断`
    };
  }

  return { valid: true, adjusted: _adjustedChange, reason: '通过' };
}

/* ───────── 七、每日/每周累计校验函数（思维链第 6 步辅助调用） ─────────
   参数：
     scope         - 'daily' 或 'weekly'
     category      - DAILY_TOTAL_LIMITS / WEEKLY_TOTAL_LIMITS 的 key
     variable_name - 具体变量名（或 'any_skill'/'any_stat' 通配）
     accumulated   - 已累计的变化量
   返回：
     { within_limit, remaining, reason } */
function validateAccumulation(scope, category, variable_name, accumulated) {
  const _table = (scope === 'daily') ? DAILY_TOTAL_LIMITS : WEEKLY_TOTAL_LIMITS;
  const _cat = _table[category];
  if (!_cat || !_cat[variable_name]) {
    return { within_limit: true, remaining: null, reason: '无累计限制' };
  }
  const _limit = _cat[variable_name];

  if (accumulated > _limit.max_increase) {
    return {
      within_limit: false,
      remaining: 0,
      reason: `${scope}累计正向超限：${variable_name} 已累计 ${accumulated} > 上限 ${_limit.max_increase}`
    };
  }
  if (accumulated < _limit.max_decrease) {
    return {
      within_limit: false,
      remaining: 0,
      reason: `${scope}累计负向超限：${variable_name} 已累计 ${accumulated} < 下限 ${_limit.max_decrease}`
    };
  }
  const _remaining = Math.min(_limit.max_increase - accumulated, accumulated - _limit.max_decrease);
  return {
    within_limit: true,
    remaining: _remaining,
    reason: '通过'
  };
}
