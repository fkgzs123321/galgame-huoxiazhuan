this.calcSalary = function calcSalary(jobName, difficulty, lcgTier) {
  const base = BASE_SALARY[jobName] || 0;
  const diffMult = (DIFFICULTY_MODIFIERS[difficulty] || DIFFICULTY_MODIFIERS['普通']).money_mult;
  const tierMult = (LCG_TIER_MODIFIER[lcgTier] || LCG_TIER_MODIFIER['green']).mult;
  return Math.round(base * diffMult * tierMult);
}
