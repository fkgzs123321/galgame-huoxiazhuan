  var _d = Number(diff) || 0;
  if (_d >= 30)  return TIER_EFFECTS.critical_success;
  if (_d >= 10)  return TIER_EFFECTS.success;
  if (_d >= -10) return TIER_EFFECTS.normal;
  if (_d >= -30) return TIER_EFFECTS.failure;
  return TIER_EFFECTS.critical_failure;
}

/* ════════════════════════════════════════════════════════════
   九、最终判定（能力值 vs 要求值对比 + LCG 骰子）
   calcFinalResult(action_type, heroine_id)
   任何恋爱事件判定的统一入口，返回完整结果对象。
   - action_type：见 ACTION_TYPES 的 key
   - heroine_id：当前对话女角 ID（0=无对话，仅用于日志）
   注意：_r() 由 LCG骰子/引擎.txt 提供，本函数不重新实现骰子。
   ════════════════════════════════════════════════════════════ */
this.calcFinalResult = function calcFinalResult(action_type, heroine_id) {
  var _actionType = action_type || 'confess';
  var _hid_local  = Number(heroine_id) || 0;
  var _actionInfo  = ACTION_TYPES[_actionType];
  if (!_actionInfo) {
    return { success: false, error: '未知动作类型: ' + _actionType, roll: -1, power: 0, requirement: 0 };
  }

  /* 1. 主角能力值（按女角偏好加权平均） */
  var _power = calcPlayerPower(_actionType, _hid_local);

  /* 2. 女角要求值（基础 + 阶段 + 难度 + 女角特质） */
  var _reqObj = calcHeroineRequirement(_actionType, _hid_local);
  var _req = _reqObj.value;

  /* 3. 差值 = 能力值 - 要求值 */
  var _diff = _power - _req;

  /* 4. 成功率 = 50% + 差值（封顶 5%-95%） */
  var _successRate = Math.max(5, Math.min(95, 50 + _diff));

  /* 5. LCG 骰子（确定性随机，由 LCG骰子/引擎.txt 的 _r() 提供） */
  var _diceRoll = _r() * 100;  /* 0~99.99 */
  var _success = _diceRoll < _successRate;

  /* 6. 5 档分档（基于差值，非随机偏移） */
  var _tier = classifyDiffResult(_diff);

  return {
    action_type: _actionType,
    action_name: _actionInfo.name,
    heroine_id: _hid_local,
    success: _success,
    tier: _tier.key,
    tier_label: _tier.label,
    effect_mult: _tier.effect_mult,
    tier_desc: _tier.desc,
    power: Math.round(_power * 100) / 100,
    requirement: Math.round(_req * 100) / 100,
    diff: Math.round(_diff * 100) / 100,
    success_rate: Math.round(_successRate * 100) / 100,
    roll: Math.round(_diceRoll * 100) / 100,
    breakdown: {
      power_value: Math.round(_power * 100) / 100,
      req_base: _reqObj.base,
      req_stage: _reqObj.stage,
      req_stage_mod: _reqObj.stage_mod,
      req_difficulty_mod: _reqObj.difficulty_mod,
      req_heroine_diff: _reqObj.heroine_diff,
      tier_examples: _tier.examples
    }
  };
}
