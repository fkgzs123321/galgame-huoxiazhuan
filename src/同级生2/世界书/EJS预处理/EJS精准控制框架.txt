/* 关系阶段等级映射（用于 >= 比较）

const EJF_STAGE_RANK = {
  '决裂': -5, '仇视': -4, '厌恶': -3, '冷漠': -2, '疏离': -1,
  '初识': 0, '熟悉': 1, '暧昧': 2, '心动': 3, '亲密': 4

function ejf_isStageOrHigher(current_stage, required_stage) {
  const _cur = EJF_STAGE_RANK[current_stage] !== undefined ? EJF_STAGE_RANK[current_stage] : 0;
  const _req = EJF_STAGE_RANK[required_stage] !== undefined ? EJF_STAGE_RANK[required_stage] : 0;
  return _cur >= _req;
}

/* 条件评估器：解析 condition 字符串并返回布尔值 */
function ejf_evaluateCondition(condition, context) {
  if (condition === 'always') return true;

  /* day_count == N */
  if (condition.startsWith('day_count == ')) {
    return context.day_count === Number(condition.split('== ')[1]);
  }
  /* current_heroine_id > N */
  if (condition.startsWith('current_heroine_id > ')) {
    return context.current_heroine_id > Number(condition.split('> ')[1]);
  }
  /* relationship_stage >= 阶段名 */
  if (condition.startsWith('relationship_stage >= ')) {
    const _stage = condition.split('>= ')[1];
    return ejf_isStageOrHigher(context.relationship_stage, _stage);
  }
  /* affection >= N */
  if (condition.startsWith('affection >= ')) {
    return context.affection >= Number(condition.split('>= ')[1]);
  }
  /* location.includes("关键词") */
  if (condition.startsWith('location.includes(')) {
    const _match = condition.match(/"([^"]+)"/);
    if (!_match) return false;
    return context.location.includes(_match[1]);
  }
  return false;
}

   ════════════════════════════════════════════════════════════ */
const TOKEN_BUDGET = {
  total_per_turn: 30000,
  breakdown: {
    system_prompt: 5000,
    ejs_preprocessing: 15000,
    heroine_data: 3000,
    scene_description: 2000,
    ai_response: 5000
  }
};

/* Token 超预算时自动精简策略 */
function ejf_enforceTokenBudget(load_result) {
  const _estimated = load_result.estimated_tokens;
  const _budget = TOKEN_BUDGET.breakdown.ejs_preprocessing;

  if (_estimated <= _budget) {
    return { action: 'none', message: '预算内' };
  }

  /* 超预算：按优先级移除低优先级模块 */
  const _priority = {
    '高': ['全局规则核心', 'LCG骰子', 'D0控制器', '防口胡机制'],
    '中': ['角色性格控制器', '恋爱互动系统', '当前女角性格', '当前女角NSFW反差'],
    '低': ['个人反应属性', '阶段联动框架', '变量变化幅度控制', '属性系统']
  };

  const _removable = load_result.loaded_modules.filter(m =>
    _priority['低'].includes(m.name) || _priority['中'].includes(m.name)
  );

  return {
    action: 'reduce',
    message: '超预算，建议移除低优先级模块',
    removable_modules: _removable.map(m => m.name),
    current_estimate: _estimated,
    budget: _budget,
    new_estimate: _budget
  };
}
