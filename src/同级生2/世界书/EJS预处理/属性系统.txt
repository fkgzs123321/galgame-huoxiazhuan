/* ─── 辅助函数：根据属性值获取分档 ─── */
this.getStatStage = function getStatStage(statName, value) {
  const v = Number(value) || 0;
  const stat = PLAYER_STATS[statName];
  if (!stat || !stat.thresholds) return null;
  for (const key in stat.thresholds) {
    const t = stat.thresholds[key];
    if (v >= t.range[0] && v <= t.range[1]) return { name: key, ...t };
  }
  return null;
}

}

/* ─── 辅助函数：计算告白成功率 ─── */
this.calcConfessionRate = function calcConfessionRate(stats) {
  const base = ROMANCE_INFLUENCE.confession_success_rate.base;
  const w = ROMANCE_INFLUENCE.confession_success_rate.weights;
  const charm = Number(stats.charm) || 0;
  const social = Number(stats.social) || 0;
  const perception = Number(stats.perception) || 0;
  const willpower = Number(stats.willpower) || 0;
  return Math.min(100, Math.round(base + charm * w.charm + social * w.social + perception * w.perception + willpower * w.willpower));
}

/* ─── 辅助函数：计算初印象加成 ─── */
this.calcFirstImpression = function calcFirstImpression(charm) {
  const c = Number(charm) || 0;
  return Math.round(c / 10 - 5);
}
