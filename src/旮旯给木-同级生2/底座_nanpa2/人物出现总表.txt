/* 从天数推导星期（12-22 是周五）—— `时间.星期` 已删，改为算出来 */
const _WEEKDAY_NAMES = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];
function getWeekday(day) {
  return _WEEKDAY_NAMES[(5 + (day - 1)) % 7];   // 5 = 周五
}

/* 某地点此刻有谁 */
function getAvailable(day, slot, place) {
  let base = ((SCHEDULE[slot] || {})[place] || []).slice();
  const ov = DATE_OVERRIDES[day] || {};
  const locked = ov.locked || [];
  base = base.filter(id => locked.indexOf(id) === -1);
  /* 周间休假（WEEKDAY_OFFS 目前为空，结构先接好） */
  const off = WEEKDAY_OFFS[getWeekday(day)];
  if (off && off.length) base = base.filter(id => off.indexOf(id) === -1);
  return base;
}

/* 当日可遇全集（供 AI 参考；不含未解锁的隐藏线） */
function getDayHeroines(day) {
  const ov = DATE_OVERRIDES[day] || {};
  const locked = ov.locked || [];
  const all = new Set();
  Object.keys(SCHEDULE).forEach(slot => {
    Object.keys(SCHEDULE[slot]).forEach(place => {
      getAvailable(day, slot, place).forEach(id => all.add(id));
    });
  });
  (ov.new_unlock || []).forEach(id => all.add(id));
  return Array.from(all).filter(id => locked.indexOf(id) === -1).sort((a, b) => a - b);
}

const _today = getDayHeroines(_dayCount);
