    var r = Number(rounds) || 0;
    var diffMap = { '简单': 0.5, '普通': 1.0, '困难': 1.5, '噩梦': 2.0 };
    var coeff = diffMap[difficulty] != null ? diffMap[difficulty] : 1.0;
    var gain = r * (1 + coeff);
    if (isBoss) gain = gain * 1.5;
    return Math.floor(gain);
  },

    var table = this.法术解锁表[jadeName];
    if (!table || stage.等级 === 0) return [];
    var spells = [];
    for (var lv = 1; lv <= stage.等级; lv++) {
      if (table[lv]) spells = spells.concat(table[lv]);
    }
    return spells;
  },

  isJadeMastered: function(ap) {
    return this.getApStage(ap).等级 >= 1;
  }
