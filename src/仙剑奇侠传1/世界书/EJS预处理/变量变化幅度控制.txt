  limits: {

  /* ───── 2. 禁止变化项（一律拒绝） ───── */
  forbidden: {
    '主角姓名':         '开局固定，不可修改',
    '性别':             '开局固定，不可修改',
    '年龄':             '开局固定，仅由剧情触发器自然增长',
    '章节进度':         '仅由章节触发器修改（完成关键事件后推进）',

    '已死亡角色状态':   '不可复活，除非触发复活事件'
  },

  /* ───── 3. 异常检测函数 ───── */
  check: function (name, before, after, type) {
    /* 禁止变化项直接拒绝 */
    if (this.forbidden.hasOwnProperty(name)) {
      return { ok: false, level: 'reject', msg: '【拒绝·禁止变化】' + name + ' 不可修改（' + this.forbidden[name] + '）' };
    }
    const rule = this.limits[name];
    if (!rule) {
      return { ok: true, level: 'skip', msg: '【跳过】' + name + ' 无明确限制规则' };
    }
    const diff = Number(after) - Number(before);
    const abs  = Math.abs(diff);
    let limit  = rule[type];
    if (limit === undefined) {
      /* 未指定事件类型，取该变量所有阈值中的最大值作为兜底 */
      const vals = Object.keys(rule).map(function (k) { return rule[k]; });
      limit = Math.max.apply(null, vals);
    }
    if (abs > limit) {
      return { ok: false, level: 'warn', msg: '【警告·超限】' + name + ' 变化 ' + (diff > 0 ? '+' : '') + diff + '，阈值 ≤' + limit + '（事件：' + type + '）' };
    }
    return { ok: true, level: 'pass', msg: '【通过】' + name + ' 变化 ' + (diff > 0 ? '+' : '') + diff + '（阈值 ≤' + limit + '，' + type + '）' };
  },

  /* ───── 4. 批量校验（处理本轮变化日志） ───── */
  validate: function (log) {
    if (!Array.isArray(log) || log.length === 0) {
      return { results: [], warns: [], rejects: [] };
    }
    const results = log.map(function (e) {
      return VALIDATE_CHANGE.check(e.name, Number(e.before) || 0, Number(e.after) || 0, e.type);
    });
    const warns   = results.filter(function (r) { return !r.ok && r.level === 'warn'; });
    const rejects = results.filter(function (r) { return !r.ok && r.level === 'reject'; });
    return { results: results, warns: warns, rejects: rejects };
  }
};

define('VALIDATE_CHANGE', VALIDATE_CHANGE);
