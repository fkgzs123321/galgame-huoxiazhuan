  getEventsByChapter: function (chapterIndex) {
    var key = 'chapter_' + chapterIndex;
    return WORLDLINE_PROPOSAL.canonical_events[key] || [];
  },
  getEventById: function (eventId) {
    var groups = WORLDLINE_PROPOSAL.canonical_events;
    for (var g in groups) {
      if (!groups.hasOwnProperty(g)) continue;
      var arr = groups[g];
      for (var i = 0; i < arr.length; i++) {
        if (arr[i].id === eventId) return arr[i];
      }
    }
    return null;
  },
  getEventChapter: function (eventId) {
    var groups = WORLDLINE_PROPOSAL.canonical_events;
    for (var g in groups) {
      if (!groups.hasOwnProperty(g)) continue;
      var arr = groups[g];
      for (var i = 0; i < arr.length; i++) {
        if (arr[i].id === eventId) {
          if (g === 'timeline') return 0;
          return Number(g.replace('chapter_', ''));
        }
      }
    }
    return -1;
  },
  isEventRewritten: function (eventId) {
    if (!Array.isArray(_wlRewrittenEvents)) return false;
    return _wlRewrittenEvents.indexOf(eventId) !== -1;
  },
  getProposableEvents: function (currentChapter) {
    var result = [];
    var groups = WORLDLINE_PROPOSAL.canonical_events;
    for (var g in groups) {
      if (!groups.hasOwnProperty(g)) continue;
      var chIdx = (g === 'timeline') ? 0 : Number(g.replace('chapter_', ''));
      if (chIdx > currentChapter) continue;
      var arr = groups[g];
      for (var i = 0; i < arr.length; i++) {
        if (!WORLDLINE_PROPOSAL.isEventRewritten(arr[i].id)) {
          result.push({
            id: arr[i].id,
            name: arr[i].name,
            chapter: (g === 'timeline') ? '历史背景' : ('第' + chIdx + '章'),
            desc: arr[i].desc
          });
        }
      }
    }
    return result;
  },
  countTotalEvents: function () {
    var total = 0;
    var groups = WORLDLINE_PROPOSAL.canonical_events;
    for (var g in groups) {
      if (!groups.hasOwnProperty(g)) continue;
      total += groups[g].length;
    }
    return total;
  }
};
