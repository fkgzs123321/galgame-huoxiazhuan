/* 条件评估器：解析 condition 字符串并返回布尔值 */
function ejf_evaluateCondition(condition, context) {
  if (condition === 'always') return true;

  /* scene_characters.length > 0 */
  if (condition === 'scene_characters.length > 0') {
    return context.scene_characters.length > 0;
  }
  /* party_names.length > 0 */
  if (condition === 'party_names.length > 0') {
    return context.party_names.length > 0;
  }
  /* current_dialogue_char > 0 */
  if (condition === 'current_dialogue_char > 0') {
    return context.current_dialogue_char > 0;
  }
  /* current_dialogue_char > N */
  if (condition.startsWith('current_dialogue_char > ')) {
    return context.current_dialogue_char > Number(condition.split('> ')[1]);
  }
  /* location.includes("关键词") */
  if (condition.startsWith('location.includes(')) {
    const _match = condition.match(/"([^"]+)"/);
    if (!_match) return false;
    return context.location.includes(_match[1]);

  /* chapter == 章节名 */
  if (condition.startsWith('chapter == ')) {
    return context.chapter === condition.split('== ')[1];
  }
  return false;
}

const TOKEN_BUDGET = {
  total_per_turn: 30000,
  ejs_preprocessing_limit: 15000,
  breakdown: {
    system_prompt: 5000,
    ejs_preprocessing: 15000,
    character_data: 3000,
    scene_description: 2000,
    ai_response: 5000
  }

/* Token 超预算时自动精简策略 */
function ejf_enforceTokenBudget(load_result) {
  const _estimated = load_result.estimated_tokens;
  const _budget = TOKEN_BUDGET.ejs_preprocessing_limit;

  if (_estimated <= _budget) {
    return { action: 'none', message: '预算内' };
  }

  /* 超预算：按优先级移除低优先级模块 */
  const _priority = {
    '高': ['全局规则核心', 'LCG骰子引擎', '主调度控制器', '战斗指令调度器', '五灵法术体系', '剑法绝技系统'],
    '中': ['灵玉镶嵌系统', '蛊毒炼丹系统', '角色性格控制器', '当前对话人物4件套', '队伍成员4件套', '当前场景人物4件套'],
    '低': ['阶段联动框架', '女娲血脉觉醒表', '变量变化幅度控制', '世界观场景控制器', '当前地点', '关系网基础信息', '队伍成员基础信息']
  };

  const _removable = load_result.loaded_modules.filter(function (m) {
    return _priority['低'].indexOf(m.name) !== -1 || _priority['中'].indexOf(m.name) !== -1;
  });

  return {
    action: 'reduce',
    message: '超预算，建议移除低优先级模块',
    removable_modules: _removable.map(function (m) { return m.name; }),
    current_estimate: _estimated,
    budget: _budget,
    new_estimate: _budget
  };
}
