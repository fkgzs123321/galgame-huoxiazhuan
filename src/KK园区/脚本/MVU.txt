/* ========= 工具函数 ========= */
function hasMeaningfulStatData(stat) {
  return stat && typeof stat === 'object' && !Array.isArray(stat) && Object.keys(stat).length > 0;
}

function pickStatData(source) {
  if (!source || typeof source !== 'object') return null;
  var stat = source.stat_data;
  return hasMeaningfulStatData(stat) ? stat : null;
}

function cloneObj(obj) {
  if (!obj || typeof obj !== 'object') return {};
  try { return JSON.parse(JSON.stringify(obj)); } catch (e) { return {}; }
}

/* ========= deepMergeStat 递归合并函数 ========= */
/* 将 source 深合并到 target，返回新对象（不修改原对象） */
function deepMergeStat(target, source) {
  if (!source || typeof source !== 'object') return target;
  var out = cloneObj(target);
  for (var k in source) {
    if (!Object.prototype.hasOwnProperty.call(source, k)) continue;
    var sv = source[k];
    var dv = out[k];
    if (sv && typeof sv === 'object' && !Array.isArray(sv) &&
        dv && typeof dv === 'object' && !Array.isArray(dv)) {
      out[k] = deepMergeStat(dv, sv);
    } else {
      out[k] = sv;
    }
  }
  return out;
}

/* ========= readStatData 读取 MVU 变量 ========= */
/* 优先级：MVU API → getVariables → embedded(首楼兜底) → DEFAULT_STAT */
/* opt: { message_id } 可选，默认读取最新楼层 */
function readStatData(opt) {
  var options = opt || {};
  var messageId = options.message_id;
  var query = { type: 'message' };
  if (messageId !== undefined && messageId !== null) {
    query.message_id = messageId;
  } else {
    query.message_id = 'latest';
  }

  /* 1. 尝试 MVU API */
  try {
    if (typeof Mvu !== 'undefined' && Mvu && typeof Mvu.getMvuData === 'function') {
      var mvuData = Mvu.getMvuData(query);
      var stat = pickStatData(mvuData);
      if (stat) return deepMergeStat(DEFAULT_STAT, stat);
    }
  } catch (e) {
    console.warn('[KK园区 MVU] MVU API 读取失败:', e);
  }

  /* 2. 尝试 getVariables */
  try {
    if (typeof getVariables === 'function') {
      var vars = getVariables(query);
      var stat2 = pickStatData(vars);
      if (stat2) return deepMergeStat(DEFAULT_STAT, stat2);
    }
  } catch (e) {
    console.warn('[KK园区 MVU] getVariables 读取失败:', e);
  }

  /* 3. 尝试首楼兜底（embedded：读取第 0 楼的 MVU 变量） */
  try {
    if (typeof Mvu !== 'undefined' && Mvu && typeof Mvu.getMvuData === 'function') {
      var embedded = Mvu.getMvuData({ type: 'message', message_id: 0 });
      var stat3 = pickStatData(embedded);
      if (stat3) return deepMergeStat(DEFAULT_STAT, stat3);
    }
  } catch (e) {
    console.warn('[KK园区 MVU] 首楼兜底读取失败:', e);
  }

  /* 4. 返回 DEFAULT_STAT 兜底 */
  return cloneObj(DEFAULT_STAT);
}

/* ========= writeStatData 写入 MVU 变量 ========= */
/* 通过 setVariables / insertOrAssignVariables 写入 */
/* opt: { type, message_id } 可选，默认写入最新楼层 */
function writeStatData(data, opt) {
  var options = opt || {};
  var type = options.type || 'message';
  if (!data || typeof data !== 'object') {
    return Promise.resolve(false);
  }

  /* 优先使用 Mvu.replaceMvuData */
  try {
    if (typeof Mvu !== 'undefined' && Mvu && typeof Mvu.replaceMvuData === 'function') {
      var query = { type: type };
      if (options.message_id !== undefined && options.message_id !== null) {
        query.message_id = options.message_id;
      } else {
        query.message_id = 'latest';
      }
      return Mvu.replaceMvuData({ stat_data: data }, query).then(function () {
        return true;
      }).catch(function (e) {
        console.warn('[KK园区 MVU] Mvu.replaceMvuData 写入失败:', e);
        return writeViaSetVariables(data, type);
      });
    }
  } catch (e) {
    console.warn('[KK园区 MVU] Mvu API 写入异常:', e);
  }

  /* 回退到 setVariables / insertOrAssignVariables */
  return writeViaSetVariables(data, type);
}

/* 通过 setVariables 或 insertOrAssignVariables 写入变量 */
function writeViaSetVariables(data, type) {
  try {
    if (typeof setVariables === 'function') {
      var result = setVariables({ stat_data: data }, { type: type });
      if (result && typeof result.then === 'function') {
        return result.then(function () { return true; }).catch(function (e) {
          console.warn('[KK园区 MVU] setVariables 写入失败:', e);
          return false;
        });
      }
      return Promise.resolve(true);
    }
  } catch (e) {
    console.warn('[KK园区 MVU] setVariables 写入异常:', e);
  }

  try {
    if (typeof insertOrAssignVariables === 'function') {
      return insertOrAssignVariables({ stat_data: data }, { type: type }).then(function () {
        return true;
      }).catch(function (e) {
        console.warn('[KK园区 MVU] insertOrAssignVariables 写入失败:', e);
        return false;
      });
    }
  } catch (e) {
    console.warn('[KK园区 MVU] insertOrAssignVariables 写入异常:', e);
  }

  console.warn('[KK园区 MVU] 无可用变量写入接口');
  return Promise.resolve(false);
}

/* ========= lcgRoll LCG 确定性骰子 ========= */
/* 基于种子和 message_id 生成确定性随机数，返回 1~100 的整数 */
/* seed: LCG种子值（来自 MVU 变量 stat_data.通用.LCG种子） */
/* mod:  消息楼层 ID 或其他修饰因子（确保不同楼层结果不同） */
function lcgRoll(seed, mod) {
  var s = ((Number(seed) || 0) * 4 + (Number(mod) || 0) + 7919) | 0;
  s = (s * 1664525 + 1013904223) | 0;
  var r = (s >>> 0) / 4294967296;
  return Math.floor(r * 100) + 1;
}

/* ========= refreshStatusBar 状态栏刷新 ========= */
/* 尝试多种方式触发状态栏刷新 */
function refreshStatusBar() {
  try {
    /* 尝试调用全局 refresh 函数（状态栏 HTML 中定义） */
    if (typeof window.refresh === 'function') {
      window.refresh();
      return;
    }
    /* 尝试通过 postMessage 通知 iframe 状态栏 */
    var iframes = document.querySelectorAll('iframe');
    for (var i = 0; i < iframes.length; i++) {
      try {
        iframes[i].contentWindow.postMessage({ type: 'kk_statusbar_refresh' }, '*');
      } catch (e) {}
    }
    /* 也尝试通知父窗口（脚本可能运行在 iframe 内） */
    if (window.parent && window.parent !== window) {
      try {
        window.parent.postMessage({ type: 'kk_statusbar_refresh' }, '*');
      } catch (e) {}
    }
  } catch (e) {
    console.warn('[KK园区 MVU] 刷新状态栏失败:', e);
  }
}

/* ========= autoRefresh 延迟刷新封装 ========= */
/* 800ms 延迟后调用 refreshStatusBar，给 MVU 扩展时间完成数据持久化 */
function autoRefresh() {
  setTimeout(refreshStatusBar, 800);
}

/* ========= 事件监听 ========= */
function setupEventListeners() {
  /* 酒馆助手事件监听 */
  try {
    if (typeof eventOn === 'function') {
      /* 生成结束事件：AI 回复生成完毕 */
      eventOn('GENERATION_ENDED', autoRefresh);
      /* 消息编辑事件：用户或 AI 编辑已有消息 */
      eventOn('MESSAGE_EDITED', autoRefresh);
      /* MVU 变量更新结束事件：MagVarUpdate 完成变量解析 */
      eventOn('mag_variable_update_ended', autoRefresh);
    }
  } catch (e) {
    console.warn('[KK园区 MVU] 事件监听注册失败:', e);
  }

  /* MVU VARIABLE_UPDATE_ENDED 事件（如果 MVU 已初始化） */
  try {
    if (typeof Mvu !== 'undefined' && Mvu && Mvu.events &&
        Mvu.events.VARIABLE_UPDATE_ENDED && typeof eventOn === 'function') {
      eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, autoRefresh);
    }
  } catch (e) {}

  /* iframe postMessage 兜底（用于嵌入状态栏场景） */
  try {
    window.addEventListener('message', function (e) {
      try {
        var data = e.data;
        var t = (data && data.type) || (typeof data === 'string' ? data : '') || '';
        if (typeof t === 'string' &&
            (t.indexOf('generation_ended') >= 0 ||
             t.indexOf('message_edit') >= 0 ||
             t.indexOf('mag_variable_update') >= 0 ||
             t.indexOf('kk_statusbar_refresh') >= 0)) {
          autoRefresh();
        }
      } catch (x) {}
    });
  } catch (e) {}
}

/* ========= 等待 MVU 初始化 ========= */
function waitForMvuInit(callback) {
  /* 优先使用 waitGlobalInitialized */
  if (typeof waitGlobalInitialized === 'function') {
    waitGlobalInitialized('Mvu').then(function () {
      callback(true);
    }).catch(function () {
      callback(false);
    });
    return;
  }
  /* 回退轮询等待 */
  var count = 0;
  var maxRetry = 50;
  function check() {
    if (typeof Mvu !== 'undefined' && Mvu) {
      callback(true);
    } else if (count < maxRetry) {
      count++;
      setTimeout(check, 200);
    } else {
      console.warn('[KK园区 MVU] MVU 初始化等待超时');
      callback(false);
    }
  }
  check();
}

/* ========= 初始化 ========= */
function init() {
  setupEventListeners();
  console.log('[KK园区 MVU] 脚本已加载');
}

/* 执行初始化（事件监听立即注册，不阻塞 MVU 初始化） */
init();

/* 后台等待 MVU 初始化完成后补充注册 VARIABLE_UPDATE_ENDED 事件 */
waitForMvuInit(function (ok) {
  if (ok) {
    try {
      if (typeof Mvu !== 'undefined' && Mvu && Mvu.events &&
          Mvu.events.VARIABLE_UPDATE_ENDED && typeof eventOn === 'function') {
        eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, autoRefresh);
      }
    } catch (e) {}
    console.log('[KK园区 MVU] MVU 框架已就绪');
  }
});
