const calcTimeSlot = (hour) => {
  const h = Number(hour) || 0;
  if (h >= 5 && h < 7) return '早晨';
  if (h >= 7 && h < 8) return '晨读';
  if (h >= 8 && h < 12) return '上午';
  if (h >= 12 && h < 14) return '中午';
  if (h >= 14 && h < 17) return '下午';
  if (h >= 17 && h < 19) return '傍晚';
  if (h >= 19 && h < 22) return '晚自习';
  if (h >= 22 || h < 2) return '夜里';
  return '半夜';
};

// ===== 工具：学业阶段映射（分数 → 阶段，AI 禁手写） =====
const calcAcademicStage = (score) => {
  const s = Number(score) || 0;
  if (s >= 90) return 5;
  if (s >= 75) return 4;
  if (s >= 60) return 3;
  if (s >= 40) return 2;
  return 1;
};

const calcCompositeStage = (理性, 执念, 理智, 逃避意识) => {
  const r = Number(理性) || 0;
  const o = Number(执念) || 0;
  const s = Number(理智) || 100;
  const e = Number(逃避意识) || 0;
  if (s <= 0) return 'E_崩坏期';
  if (e >= 70 && r >= 40) return 'E_清醒反扑期';
  if (o >= 70 || s <= 30) return 'D_疯狂期';
  if (o >= 40 || r <= 40) return 'C_沉沦期';
  if (o >= 15 || r <= 65) return 'B_动摇期';
  return 'A_理性期';
};
