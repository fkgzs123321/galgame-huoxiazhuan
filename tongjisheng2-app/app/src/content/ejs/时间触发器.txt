/* ───────── 四、条件校验辅助函数 ───────── */
function checkCondition(condition, ctx) {
  if (!condition) return true;
  ctx = ctx || {};

  const _flags = ctx.flags || {};
  const _money = Number(ctx.money) || 0;
  const _affection = Number(ctx.affection) || 0;
  const _heroineId = Number(ctx.heroine_id) || 0;
  const _hospitalVisits = Number(ctx.hospital_visits) || 0;
  const _location = ctx.location || '';
  const _time = ctx.time || '';

  // hospital_visits_min：市民医院累计访问次数下限
  if (typeof condition.hospital_visits_min === 'number') {
    if (_hospitalVisits < condition.hospital_visits_min) return false;
  }

  // flag_required：必须已持有的标记
  if (condition.flag_required) {
    if (!_flags[condition.flag_required]) return false;
  }

  // affection_min：当前女角好感下限
  if (typeof condition.affection_min === 'number') {
    if (_affection < condition.affection_min) return false;
  }

  // money_min：金钱下限
  if (typeof condition.money_min === 'number') {
    if (_money < condition.money_min) return false;
  }

  // heroine_id：限定女角
  if (typeof condition.heroine_id === 'number') {
    if (_heroineId !== condition.heroine_id) return false;
  }

  // location：限定地点（支持字符串与数组）
  if (condition.location) {
    if (Array.isArray(condition.location)) {
      if (!condition.location.includes(_location)) return false;
    } else if (_location !== condition.location) {
      return false;
    }
  }

  // time：限定时间（支持 "22:00+" 前缀匹配表示 ≥22:00）
  if (condition.time) {
    if (condition.time.endsWith('+')) {
      const _threshold = condition.time.slice(0, -1);
      if (_time < _threshold) return false;
    } else if (_time !== condition.time) {
      return false;
    }
  }

  return true;
}

/* ───────── 五、触发器主校验函数 ───────── */
function checkTriggers(day_count, time_slot, location, flags, extraCtx) {
  const _triggers = [];
  const _ctx = Object.assign({ flags, location }, extraCtx || {});

  // 1. 日期触发
  if (DATE_TRIGGERS[day_count]) {
    DATE_TRIGGERS[day_count].triggers.forEach(t => {
      if (!t.time_slot || t.time_slot === time_slot) {
        if (!t.condition || checkCondition(t.condition, _ctx)) {
          _triggers.push(t);
        }
      }
    });
  }

  // 2. 时段触发
  if (SLOT_TRIGGERS[time_slot]) {
    SLOT_TRIGGERS[time_slot].triggers.forEach(t => {
      if (!t.condition || checkCondition(t.condition, _ctx)) {
        _triggers.push(t);
      }
    });
  }

  // 3. 日期+时段组合触发
  const _dateSlotKey = `${day_count}_${time_slot}`;
  if (DATE_SLOT_TRIGGERS[_dateSlotKey]) {
    DATE_SLOT_TRIGGERS[_dateSlotKey].triggers.forEach(t => {
      if (!t.condition || checkCondition(t.condition, _ctx)) {
        _triggers.push(t);
      }
    });
  }

  return _triggers;
}
