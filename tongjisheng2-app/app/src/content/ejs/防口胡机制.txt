function validateAction(action_type, heroine_id, context) {
  const _result = {
    valid: true,
    violations: [],
    warnings: [],
    blocked: false,
    clamped_values: {}
  };
  
  // 1. 读取难度（移除兜底值，未初始化时返回 undefined）
  const _diff = context.difficulty;
  const _strict = DIFFICULTY_STRICTNESS[_diff] || DIFFICULTY_STRICTNESS['普通'];
  
  // 2. 校验前置flag
  if (ACTION_FLAG_REQUIREMENTS[action_type]) {
    const _req = ACTION_FLAG_REQUIREMENTS[action_type];
    
    // 校验affection
    if (_req.required_affection && context.affection < _req.required_affection) {
      _result.valid = false;
      _result.violations.push({
        rule: 'affection_too_low',
        actual: context.affection,
        required: _req.required_affection,
        desc: `${action_type}需要affection≥${_req.required_affection}，当前${context.affection}`
      });
    }
    
    // 校验关系阶段
    if (_req.required_relationship_stage) {
      const _stageOrder = ['决裂','仇视','厌恶','冷漠','疏离','初识','熟悉','暧昧','心动','亲密'];  // 10阶段新版（5负+5正），单一来源：阶段联动框架.txt.STAGE_FRAMEWORK
      const _reqIdx = _stageOrder.indexOf(_req.required_relationship_stage);
      const _curIdx = _stageOrder.indexOf(context.relationship_stage);
      if (_curIdx < _reqIdx) {
        _result.valid = false;
        _result.violations.push({
          rule: 'stage_too_low',
          actual: context.relationship_stage,
          required: _req.required_relationship_stage,
          desc: `${action_type}需要关系阶段≥${_req.required_relationship_stage}，当前${context.relationship_stage}`
        });
      }
    }
    
    // 校验flag
    if (_req.required_flags && _req.required_flags.length > 0) {
      _req.required_flags.forEach(flag => {
        if (!context.flags || context.flags[flag] !== 1) {
          _result.valid = false;
          _result.violations.push({
            rule: 'missing_flag',
            flag: flag,
            desc: `${action_type}需要flag: ${flag}`
          });
        }
      });
    }

    // 校验 MVU 变量 flag（直接读取 stat_data 变量，不走 context.flags）
    if (_req.required_mvu_flags && _req.required_mvu_flags.length > 0) {
      _req.required_mvu_flags.forEach(mvuFlag => {
        const _mvuVal = Number(getvar(mvuFlag.path, { scope: 'local', defaults: mvuFlag.defaults })) || 0;
        if (_mvuVal !== mvuFlag.expected) {
          _result.valid = false;
          _result.violations.push({
            rule: 'missing_mvu_flag',
            path: mvuFlag.path,
            actual: _mvuVal,
            required: mvuFlag.expected,
            desc: `${action_type}需要${mvuFlag.desc}（${mvuFlag.path}=${_mvuVal}，需=${mvuFlag.expected}）`
          });
        }
      });
    }
  }
  
  // 3. 难度严格度校验（地狱难度强制成功率上限）
  if (_strict.max_success_rate < 95) {
    _result.max_success_rate = _strict.max_success_rate;
    _result.min_failure_rate = _strict.min_failure_rate;
  }
  
  // 4. 阻止非法操作
  if (!_result.valid && ANTI_CHEAT_CONFIG.block_invalid_changes) {
    _result.blocked = true;
  }
  
  return _result;
}
