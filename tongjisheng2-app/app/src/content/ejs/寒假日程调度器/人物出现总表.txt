/* ─── 应用覆盖：获取当日某时段某区域的可遇女角 ID 列表 ─── */
function getAvailableHeroines(dayCount, timeSlot, region) {
  let base = (DEFAULT_SCHEDULE[timeSlot] && DEFAULT_SCHEDULE[timeSlot][region]) || [];
  const override = DATE_OVERRIDES[dayCount];
  if (override && override.locked) {
    base = base.filter(id => !override.locked.includes(id));
  }
  /* 应用周间休假：读取当日 weekday，过滤掉 WEEKDAY_OFFS 中的女角 */
  const _wd = String(getvar('stat_data.时间.星期', { scope: 'local', defaults: '周六' })) || '周六';
  if (WEEKDAY_OFFS[_wd]) {
    base = base.filter(id => !WEEKDAY_OFFS[_wd].includes(id));
  }
  return base;
}

/* ─── 当日可遇女角全集（用于 AI 加载 NPC 条目参考） ─── */
function getDayHeroines(dayCount) {
  const all = new Set();
  ['早', '上午', '下午', '晚', '深夜'].forEach(slot => {
    Object.keys(DEFAULT_SCHEDULE[slot]).forEach(region => {
      getAvailableHeroines(dayCount, slot, region).forEach(id => all.add(id));
    });
  });
  return Array.from(all).sort((a, b) => a - b);
}
