/* ════════════════════════════════════════════════════════════
   主结局判定函数（1/7 PM8:00 触发）
   ════════════════════════════════════════════════════════════ */
function determineEnding(heroine_id) {
  /* 1. 优先检查 BAD_END（不可逆，通用）*/
  const badEnd = checkBadEnd();
  if (badEnd) return badEnd;

  /* 2. 检查隐藏结局（杉本樱子 heroine_id=20）*/
  if (heroine_id === 20 && _flag('樱子线解锁') && _flag('樱子死讯') && _flag('樱子反告白')) {
    return 'TRUE_END_HIDDEN';
  }

  /* 2.5 检查泉美自动告白特殊结局（heroine_id=5，9.B.7）*/
  if (heroine_id === 5 && _aff(5) >= _dm.true_end_threshold && _flag('泉美H完成') && _flag('泉美射箭援助') && _flag('泉美台词变化day15') && !_flag('主角已告白泉美')) {
    return 'IZUMI_AUTO_CONFESSION';
  }

  /* 2.6 检查美沙反告白特殊结局（heroine_id=10，9.B.6）*/
  if (heroine_id === 10 && _aff(10) >= 50 && _aff(10) < _dm.true_end_threshold && _flag('美沙重逢') && _flag('美沙训练援助') && !_flag('主角已告白美沙')) {
    return 'MISA_REVERSED_CONFESSION';
  }

  /* 3. 检查洋子怀孕特殊结局（洋子线 heroine_id=6）*/
  if (heroine_id === 6 && _yokoPregnancyFlag === 1 && _yokoUnprotected >= 1 && (_yokoRelationship === '亲密' || _yokoRelationship === '攻略完成')) {
    return 'YOKO_PREGNANCY';
  }

  /* 4. 按 affection 分档判定 True/Good/Normal/Bad */
  const aff = _aff(heroine_id);
  if (aff >= _dm.true_end_threshold && checkHeroineFlags(heroine_id)) return 'TRUE_END';
  if (aff >= _dm.good_end_threshold) return 'GOOD_END';
  if (aff >= 30) return 'NORMAL_END';
  return 'BAD_END';
}
