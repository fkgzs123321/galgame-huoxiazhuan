/* ─── 辅助函数：根据性格特质值获取分档 ─── */
function getTraitStage(traitName, value) {
  const v = Number(value) || 0;
  const trait = PERSONALITY_TRAITS[traitName];
  if (!trait || !trait.thresholds) return null;
  for (const key in trait.thresholds) {
    const t = trait.thresholds[key];
    if (v >= t.range[0] && v <= t.range[1]) return { name: key, ...t };
  }
  return null;
}

function getNarrativeGuide(traitName, value) {
  const v = Number(value) || 0;
  const guide = PERSONALITY_NARRATIVE_GUIDE[traitName];

  if (v >= 60) return { level: 'high', ...guide.high };

}

/* ─── 辅助函数：计算性格组合原型（返回最匹配的原型） ─── */
function detectArchetype(traits) {
  const t = traits || {};
  let bestMatch = null;
  let bestScore = 0;
  for (const name in PERSONALITY_COMBINED_ARCHETYPES) {
    const arch = PERSONALITY_COMBINED_ARCHETYPES[name];
    let matched = 0;
    let total = 0;
    for (const traitName in arch.match) {
      const range = arch.match[traitName];
      const val = Number(t[traitName]) || 0;
      total++;
      if (val >= range[0] && val <= range[1]) matched++;
    }
    const score = total > 0 ? matched / total : 0;
    if (score > bestScore && score >= 0.6) {  // 至少 60% 维度匹配才算该原型
      bestScore = score;
      bestMatch = { name, match_ratio: score, ...arch };
    }
  }
  return bestMatch || { name: '无明确原型', narrative_style: 'AI 视情境综合 6 维特质描写' };
}

/* ─── 辅助函数：计算性格对告白的修正（叠在第1层之上） ─── */
function calcPersonalityConfessionModifier(traits) {
  const t = traits || {};
  const w = PERSONALITY_ROMANCE_INFLUENCE.confession_modifier.weights;
  const sum =
    (Number(t.gentleness) || 0) * w.gentleness +
    (Number(t.decisiveness) || 0) * w.decisiveness +
    (Number(t.humor) || 0) * w.humor +
    (Number(t.cautiousness) || 0) * w.cautiousness +
    (Number(t.extraversion) || 0) * w.extraversion +
    (Number(t.rationality) || 0) * w.rationality;
  // 修正总和上限 ±10%
  return Math.max(-10, Math.min(10, Math.round(sum * 10) / 10));
}
