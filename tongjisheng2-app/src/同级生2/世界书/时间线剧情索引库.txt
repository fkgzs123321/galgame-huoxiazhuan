// 查询当前日期的剧情
function queryDayStory(day_count) {
  const _dc = Number(day_count) || 1;
  const _story = TIMELINE_INDEX[_dc];
  if (!_story) {
    return {
      found: false,
      message: 'day_count ' + _dc + ' 无专属剧情（超出 1-17 范围），使用日常剧情'
    };
  }
  return {
    found: true,
    day: _dc,
    date: _story.date,
    weekday: _story.weekday,
    chapter: _story.chapter,
    story_arc: _story.story_arc,
    summary: _story.summary,
    key_events: _story.key_events,
    available_heroines: _story.available_heroines || [],  /* 运行时从 寒假日程调度器/人物出现总表.DEFAULT_SCHEDULE 查询 */
    available_locations: _story.available_locations,
    locked_heroines: _story.locked_heroines,
    new_unlock: _story.new_unlock,
    weather_default: _story.weather_default,
    flags_triggered: _story.flags_triggered,
    bad_end_risk: _story.bad_end_risk,
    notes: _story.notes,
    ending_matrix: _story.ending_matrix
  };
}

// 查询当前时段的剧情
function querySlotStory(time_slot) {
  const _slot = SLOT_STORY_INDEX[time_slot];
  if (!_slot) {
    return {
      found: false,
      message: 'time_slot ' + time_slot + ' 无效，有效值：早/上午/下午/晚/深夜'
    };
  }
  return {
    found: true,
    time_slot: time_slot,
    time_range: _slot.time_range,
    common_events: _slot.common_events,
    heroines_available: _slot.heroines_available,
    available_actions: _slot.available_actions,
    notes: _slot.notes
  };
}

// 查询当前日期+时段的剧情（组合查询）
function queryCurrentStory(day_count, time_slot) {
  const _dayStory = queryDayStory(day_count);
  const _slotStory = querySlotStory(time_slot);

  // 筛选当日此时段的专属事件
  var _currentSlotEvents = [];
  if (_dayStory.found && _dayStory.key_events) {
    _currentSlotEvents = _dayStory.key_events.filter(function(e) {
      return e.time_slot === time_slot || e.time_slot === '全天';
    });
  }

  return {
    day_story: _dayStory,
    slot_story: _slotStory,
    current_slot_events: _currentSlotEvents,
    is_forced_event: _currentSlotEvents.some(function(e) { return e.forced === true; })
  };
}
